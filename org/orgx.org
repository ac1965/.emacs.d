  (defun my/org-buffer-files ()
    "Return a list of *.org files currently visited in live buffers."
    (cl-loop for buf in (buffer-list)
             for file = (buffer-file-name buf)
             when (and file (string-match-p "\\.org\\'" file))
             collect file))

  ;; org-agenda-files のフェーズ 1（フォールバック）。
  ;;
  ;; Org を遅延ロードするため、この設定は leaf org の :config に置いてはならない。
  ;; :config は (with-eval-after-load 'org ...) へ包まれるため、org が実際にロード
  ;; されるのは after-init より後（.org を開いた時点など）になる。そこでフォールバックを
  ;; 代入すると、after-init-hook のフェーズ 2 が構築した完全な一覧を、後から
  ;; inbox.org 1 件で上書きしてしまう。
  ;;
  ;; したがってフェーズ 1 はトップレベルに置き、org-directory ではなく my:d:org を
  ;; 参照する（org 未ロードでも評価できる）。順序は次のとおりである。
  ;;
  ;;   1. モジュールのロード時（ここ）      : inbox.org のみ
  ;;   2. after-init-hook（フェーズ 2）     : orgx-agenda-build の完全な一覧で置換
  ;;   3. org.el のロード時                 : (defcustom org-agenda-files ...) は
  ;;                                          束縛済みの値を上書きしない
  (setq org-agenda-files
        (list (expand-file-name "inbox.org" my:d:org)))

  ;; Org は Emacs 組み込みを使う（Emacs 32.0.50 同梱: Org 9.8.10）。
  ;; したがって :straight nil はコーディング規則 3 のとおり「組み込み」の宣言であり、
  ;; 実態と一致する。init.el が straight 版 Org を導入していた頃は、この宣言が
  ;; 実態と食い違っていた（実際には straight 版がロードされていた）。
  ;;
  ;; :require t は付けない（Org を遅延ロードするため）。
  ;; - :custom は Custom が値を先行登録するため、org.el がロードされる前に宣言しても
  ;;   defcustom の評価時に正しく反映される。
  ;; - :config は leaf により (with-eval-after-load 'org ...) へ包まれるため、
  ;;   org-directory を参照するフォールバックやキャプチャテンプレートの設定は、
  ;;   org が実際にロードされた時点で確実に実行される。
  ;; - フェーズ 2 の orgx-core--rebuild-agenda-files は after-init-hook で走り、
  ;;   orgx-agenda-build（my:d:org を見る。org-directory には依存しない）から
  ;;   完全な一覧を得て org-agenda-files を setq する。org が未ロードでも動作し、
  ;;   後から org.el が (defcustom org-agenda-files ...) を評価しても、既に束縛済みの
  ;;   値は上書きされない。
  ;; このリポジトリの Org 文書（README.org と org/*.org）でだけ、ソースブロックの字下げを保つ。
  ;;
  ;; 既定（org-src-preserve-indentation が nil）では、C-c ' (org-edit-special) は
  ;; 開くときにブロックの最小の字下げを取り除き、戻るときに
  ;; org-edit-src-content-indentation（既定 2）桁を全行へ足し直す。このため
  ;; 桁 0 の行（dot ブロック、docstring の継続行）は戻るたびに 2 桁ずれる。
  ;; t にすると開閉とも字下げに触れないので、往復しても変わらない。
  ;;
  ;; グローバルには設定しない。他の Org ファイルの編集まで変わるため。
  ;; make tangle は init を読まないバッチで走るので影響を受けない（字下げの除去は
  ;; 従来どおり働く）。一方、対話的な tangle（C-c C-v t）はこのバッファローカル値を
  ;; 見て字下げを除去せず、書き出した .el 全体が 2 桁深くなる。そこで下の :around
  ;; advice が tangle の間だけ nil へ戻し、make tangle と同じ字下げにそろえる。
  ;; ファイルローカル変数は安全な値として登録されておらず確認を求められるため、
  ;; フックで束縛する。
  (defun orgx-core-source-org-p (&optional file)
    "Return non-nil when FILE is a literate source of this repository.
  FILE defaults to the current buffer's file.  The sources are README.org,
  and org/*.org under `user-emacs-directory'."
    (let ((file (or file buffer-file-name)))
      (when (and file (string-suffix-p ".org" file))
        (let ((rel (file-relative-name (file-truename file)
                                       (file-truename user-emacs-directory))))
          (and (or (equal rel "README.org")
                   (string-match-p "\\`org/[^/]+\\.org\\'" rel))
               t)))))

  (defun orgx-core--preserve-readme-src-indentation ()
    "Preserve source block indentation in this repository's Org sources."
    (when (orgx-core-source-org-p)
      (setq-local org-src-preserve-indentation t)))

  ;; 下の let を動的束縛にするため、特殊変数であることをコンパイラへ伝える
  ;; （Org を読み込まずにコンパイルされても字句的束縛にならない）。
  (defvar org-src-preserve-indentation)

  (defun orgx-core--tangle-strip-indentation (fn &rest args)
    "Call FN with ARGS while `org-src-preserve-indentation' is nil.
  Tangling from the Org sources must strip the block indentation just as
  `make tangle' does; the buffer-local t is only for editing."
    (let ((org-src-preserve-indentation nil))
      (apply fn args)))

  (add-hook 'org-mode-hook #'orgx-core--preserve-readme-src-indentation)
  (advice-add 'org-babel-tangle :around #'orgx-core--tangle-strip-indentation)

    :straight nil
    :custom
    ((org-directory . my:d:org)
     (org-default-notes-file . "notes.org")
     (org-log-done . 'time)
     (org-support-shift-select . t)
     (org-return-follows-link . t))
    :config
    ;; （例外）org-todo-keywords、org-refile-targets、org-capture-templates は
    ;; defcustom だが、その値には実行時評価の式（expand-file-name を伴うバッククォート、
    ;; (function ...) による関数参照）が含まれる。これらは値が完全に評価される前に
    ;; setopt の型チェッカが誤った警告を出す原因となる。したがってこの 3 つの代入には
    ;; setq が正しい。同じ理由により :custom や :pre-setq へ移すこともできない。
    ;; コーディング規則 6 を参照。
    ;;
    (setq org-todo-keywords
          '((sequence "TODO(t)" "SOMEDAY(s)" "WAITING(w)" "|" "DONE(d)" "CANCELED(c@)")))
    (setq org-refile-targets
          '((nil :maxlevel . 3)
            (my/org-buffer-files :maxlevel . 1)
            (org-agenda-files :maxlevel . 3)))
    ;; ブログ記事のテンプレート "b" は、orgx-hugo.el が org のロード後に追加する。
    ;; my:f:capture-blog-file（上の defvar）の個人設定による上書きは、
    ;; orgx-core のロード前に personal/<user>.el で行うこと。
    (setq org-capture-templates
          `(("t" "Todo" entry (file+headline ,(expand-file-name "gtd.org" org-directory) "Inbox")
             "* TODO %?\n %i\n %a")
            ("n" "Note" entry (file+headline ,(expand-file-name "notes.org" org-directory) "Notes")
             "* %?\nEntered on %U\n %i\n %a")
            ("j" "Journal" entry (function org-journal-find-location)
             "* %(format-time-string org-journal-time-format)%^{Title}\n%i%?")
            ("m" "Meeting" entry (file ,(expand-file-name "meetings.org" org-directory))
             "* MEETING with %? :meeting:\n  %U\n  %a"))))

  ;; org-emphasis-regexp-components の既定値（PRE/POST）は半角の空白・句読点のみを
  ;; 想定しており、全角の「」『』（）、。・ 等は含まれない。本ドキュメントのような
  ;; 和文中心の文章では、和文の直後に =code= や *bold* を続け、その直後にまた
  ;; 全角の閉じ括弧・句読点が続く書き方がごく普通に発生する
  ;; （例: 「=README.org=（設定の本体）は」）。この全角文字が PRE/POST の集合に
  ;; 無いため、Org は強調マーカーを強調として認識せず、リテラルの "=" 等の文字が
  ;; そのまま表示されてしまう（scripts/check_emphasis.py が検出する不具合の実体）。
  ;;
  ;; 対処として、和文でよく使う開き括弧類を PRE に、閉じ括弧・読点類を POST に
  ;; 追加する。org-emphasis-regexp-components の :set 関数は org-set-emph-re
  ;; であり、setopt はこれを正しく呼び出して org-emphasis-re / org-verbatim-re を
  ;; 再生成する（手動で org-set-emph-re を呼び直す必要はない）。
  ;;
  ;; ここだけ leaf の :config 本体から外し、with-eval-after-load 'org で明示的に
  ;; 遅延させている。理由: leaf の :config は org.el のロードを待たない
  ;; （org-todo-keywords 等の setq がここまで問題なく動いていたのは、setq が
  ;; 未束縛のシンボルにも新規束縛を作れる — defcustom を待つ必要がない — ためで
  ;; あって、:config 自体が遅延されているわけではない）。この customization は
  ;; org-emphasis-regexp-components の *既存の値* を nth で読んで拡張するため、
  ;; org.el が実際にロードされ defcustom の初期値が入った後でなければ
  ;; 「Symbol's value as variable is void」で失敗する。すぐ下の「アーカイブ先」と
  ;; 同じ理由・同じパターンである。
  ;;
  ;; scripts/check_emphasis.py の PRE_OK / POST_OK もこれと同じ文字集合を
  ;; 使うよう更新してある。二重管理だが、Python 側は Emacs 本体が無い環境でも
  ;; 実行できることが目的（Appendix を参照）であり、Elisp の値を実行時に import
  ;; することはできないため、手作業での同期が必要になる。
  (with-eval-after-load 'org
    (setopt org-emphasis-regexp-components
            (list (concat (nth 0 org-emphasis-regexp-components) "「『（【、。・")
                  (concat (nth 1 org-emphasis-regexp-components) "」』）】「『（【、。・")
                  (nth 2 org-emphasis-regexp-components)
                  (nth 3 org-emphasis-regexp-components)
                  (nth 4 org-emphasis-regexp-components))))

  ;; アーカイブ先
  (with-eval-after-load 'org
    (let* ((central (expand-file-name "archive.org"
                                      (or (bound-and-true-p org-directory)
                                          (bound-and-true-p my:d:org)
                                          (expand-file-name "~/org")))))
      (setopt org-archive-location (concat central "::"))))

  (defun orgx-core--rebuild-agenda-files ()
    "Phase 2: replace agenda-files fallback with the full cached list."
    (when (fboundp 'orgx-agenda-build)
      (let ((files (orgx-agenda-build)))
        (when files
          (setq org-agenda-files files)))))

  (add-hook 'after-init-hook #'orgx-core--rebuild-agenda-files)

  ;; org-persist-load-all は org（正確には org-persist.el）が自身のトップレベルで
  ;; (add-hook 'after-init-hook #'org-persist-load-all) しており、このリポジトリの
  ;; README.org 側には元々何の記述もない Org 標準の挙動である。org-persist--index
  ;; （~/.emacs.d/.var/org/persist/index.eld）に記録された全エントリ（org-element-cache
  ;; の永続化データ。my:d:org 配下、主に org-roam のノート数百件分が蓄積し得る）に
  ;; 対応するファイルを起動時に一括で開いて検証する。org-roam 配下のファイルの一部で
  ;; アクセスが不安定（Read error: Operation timed out / Resource deadlock avoided）に
  ;; なっていた環境では、この一括検証がデーモンクラッシュの引き金になっていた
  ;; （org.el が daemon 起動シーケンス中に何らかの理由で早期ロードされると、
  ;; after-init-hook 実行前に登録が間に合ってしまう）。org-element-cache は各バッファを
  ;; 実際に開いた時点で遅延生成されるため、起動時の一括ウォームアップを止めても
  ;; 機能は失われない。org-persist 自体のロードより後に確実に除去されるよう
  ;; with-eval-after-load 'org-persist の中で remove-hook する。
  (with-eval-after-load 'org-persist
    (remove-hook 'after-init-hook #'org-persist-load-all))

  (provide 'orgx-core)

  ;;; orgx/orgx-agenda.el --- キャッシュ付き Org アジェンダ・スキャナ -*- lexical-binding: t; -*-
  ;;
  ;; Copyright (c) 2021-2026
  ;; Author: YAMASHITA, Takao
  ;; License: GNU GPL v3 or later
  ;;
  ;; Category: orgx
  ;;
  ;;; Commentary:
  ;; `my:d:org' 配下を走査し、その結果を TTL 付きでキャッシュする `orgx-agenda-build'
  ;; を提供する。ロード時に走査しないよう、`orgx-core' から after-init-hook 経由で
  ;; 呼び出される。
  ;;
  ;;; Code:

  ;; eval-when-compile には (require 'cl-lib) を置かない。
  ;; cl-loop は orgx-agenda--scan 内で実行時に使われるため、下のランタイムの
  ;; require が必要かつ十分である。コンパイル時の require は冗長であり、
  ;; eval-when-compile にはコンパイル中に必要なマクロ（subr-x）だけを置くという
  ;; プロジェクト全体の規約とも矛盾する。
  (eval-when-compile (require 'subr-x))
  (require 'cl-lib)

  (defvar orgx-agenda--cache nil
    "Cache of Org agenda file paths. Nil = not yet built.")

  (defvar orgx-agenda--last-scan-time nil
    "Time of last successful scan.")

  ;; defgroup を追加し、defcustom 変数が外部の 'org-agenda グループではなく
  ;; 本モジュール自身のカスタマイズグループに属するようにした。
  ;; defcustom を所有するモジュールは、いずれも自身の defgroup を宣言しなければならない。
  (defgroup orgx-agenda nil
    "Cached Org agenda file scanning for the modular configuration."
    :group 'org)

  (defcustom orgx-agenda-max-depth 3
    "Maximum directory recursion depth for agenda file scanning."
    :type 'integer :group 'orgx-agenda)

  (defcustom orgx-agenda-exclude-dirs '("archive" ".git" "node_modules")
    "Directory names to skip during agenda scanning."
    :type '(repeat string) :group 'orgx-agenda)

  (defcustom orgx-agenda-cache-ttl 300
    "Seconds before the cache is considered stale."
    :type 'integer :group 'orgx-agenda)

  (defun orgx-agenda--should-exclude-p (dir)
    (let ((name (file-name-nondirectory (directory-file-name dir))))
      (member name orgx-agenda-exclude-dirs)))

  (defun orgx-agenda--scan (dir depth)
    ;; ディレクトリごとに directory-files を 1 回だけ呼び、その結果を .org ファイルと
    ;; 非隠しサブディレクトリに分割する。従来の実装はディレクトリごとに
    ;; directory-files を 2 回（"\.org\'" 用と "\`[^.]" 用）呼んでおり、訪問する
    ;; ディレクトリすべてでファイルシステム I/O が倍増していた。
    (when (and (file-directory-p dir) (> depth 0)
               (not (orgx-agenda--should-exclude-p dir)))
      (let* ((all     (directory-files dir t "\\`[^.]"))
             (files   (seq-filter (lambda (f) (string-match-p "\\.org\\'" f)) all))
             (subdirs (seq-filter #'file-directory-p all)))
        (append files
                (cl-loop for subdir in subdirs
                         append (orgx-agenda--scan subdir (1- depth)))))))

  (defun orgx-agenda-invalidate-cache ()
    "Force a rebuild on the next call to `orgx-agenda-build`."
    (interactive)
    (setq orgx-agenda--cache nil
          orgx-agenda--last-scan-time nil)
    (message "[orgx-agenda] cache invalidated"))

  (defun orgx-agenda-build (&optional force)
    "Return (and cache) the list of Org agenda files.
  Uses the cache unless FORCE is non-nil or the cache has expired."
    (let* ((now (float-time))
           (stale (or (null orgx-agenda--last-scan-time)
                      (> (- now orgx-agenda--last-scan-time)
                         orgx-agenda-cache-ttl))))
      (when (or force stale (null orgx-agenda--cache))
        (let* ((dir (or (bound-and-true-p my:d:org)
                        (expand-file-name "org" user-emacs-directory)))
               (files (when (file-directory-p dir)
                        (orgx-agenda--scan dir orgx-agenda-max-depth))))
          (setq orgx-agenda--cache files
                orgx-agenda--last-scan-time now))))
    orgx-agenda--cache)

  (provide 'orgx-agenda)
  ;;; orgx/orgx-agenda.el ends here

    (interactive) (org-fold-show-subtree))

    "Toggle fold state of the subtree at point."
    (interactive)
    (save-excursion

  ;;     ではなく \includesvg{*} を受け取っており、旧フィルタの正規表現が一致対象
  ;;     を持たず素通りしていた。加えて、replace-regexp-in-string に関数 REP を
  ;;     渡す場合でも、返り値に含まれる `\' はデフォルトでは literal 扱いされず

  ;;     （返り値に \linewidth のような \ を含む文字列が混ざる場合に発生）。
  ;;     下記は両方を修正済み: \includesvg{*} を直接マッチし、
  ;;     replace-regexp-in-string の LITERAL 引数を明示的に t にしている。

  (defun orgx-export--svg-to-pdf-graphics (contents backend _info)
    "Rewrite \\includesvg{FILE} to \\includegraphics{FILE.pdf} for the latex backend.
  ox-latex natively rewrites *.svg image links from \\includegraphics{FILE.svg}
  to \\includesvg{FILE} (extension already stripped) before any
  `org-export-filter-link-functions' filter sees the text — this happens
  unconditionally for any link matching `org-latex-inline-image-rules', whether
  or not the svg package (and its Inkscape dependency, deliberately avoided
  here; see the change history comment above) is actually loaded. Instead,
  *.svg figures are expected to be pre-rendered to a sibling *.pdf via
  rsvg-convert (brew install librsvg):

    for f in svg/*.svg; do rsvg-convert -f pdf -o \"${f%.svg}.pdf\" \"$f\"; done

  This filter redirects \\includesvg back to the ordinary graphicx
  \\includegraphics pipeline (no shell-escape needed) pointing at the
  pre-rendered PDF. LITERAL is passed as t explicitly: even with a function
  REP, `replace-regexp-in-string' does not treat the returned string as
  literal by default, and the replacement here can itself contain backslashes
  (e.g. \\linewidth carried over from the original options), which would
  otherwise trigger \"Invalid use of `\\' in replacement text\"."
    (when (org-export-derived-backend-p backend 'latex)
      (replace-regexp-in-string
       "\\\\includesvg\\(\\[[^]]*\\]\\)?{\\([^}]+\\)}"
       (lambda (whole)
         (string-match
          "\\\\includesvg\\(\\[[^]]*\\]\\)?{\\([^}]+\\)}" whole)
         (concat "\\includegraphics" (or (match-string 1 whole) "")
                 "{" (match-string 2 whole) ".pdf}"))
       contents
       nil t)))

  (defun orgx-export--configure-latex ()
    "Register the ltjsarticle class, route PDF export through latexmk, and
  redirect *.svg image links to their pre-rendered *.pdf sibling for the latex
  backend. `ox-latex' is required lazily here (after `org' loads) rather than
  at top level, mirroring `orgx-export--register-mermaid' above."
    (require 'ox-latex)
    (add-to-list 'org-latex-classes
                 '("ltjsarticle"
                   "\\documentclass[ja=standard]{ltjsarticle}"
                   ("\\section{%s}" . "\\section*{%s}")
                   ("\\subsection{%s}" . "\\subsection*{%s}")
                   ("\\subsubsection{%s}" . "\\subsubsection*{%s}")))
    ;; ~/.latexmkrc（LuaLaTeX-ja がデフォルト、$pdf_mode = 4）に依存関係解決と
    ;; 再コンパイル回数の判断を一任する。bibtex/makeindex も upbibtex/upmendex
    ;; として .latexmkrc 側で処理されるため、ここでは何も指定しない。
    ;; shell-escape は使わない（svg パッケージ運用を撤回したため不要。任意の
    ;; シェルコマンド実行を許す -shell-escape は使わずに済むなら使わない）。
    (setopt org-latex-pdf-process
            '("latexmk -synctex=1 -halt-on-error -output-directory=%o %f"))
    (add-to-list 'org-export-filter-link-functions
                 #'orgx-export--svg-to-pdf-graphics))

  (with-eval-after-load 'org
    (orgx-export--configure-latex))

  (provide 'orgx-export)

#+end_src

  ;;
  ;; Copyright (c) 2021-2026
  ;; Author: YAMASHITA, Takao
  ;; License: GNU GPL v3 or later
  ;;
  ;; Category: org
  ;;
  ;;; Commentary:
  ;;
  ;; 保存した Org 文書（README.org と org/*.org）を `org-babel-tangle' で自動 tangle する。
  ;;
  ;; 本ファイルの責務:
  ;; - `orgx-core-source-org-p' が真のファイル（README.org と org/*.org）のみを
  ;;   対象とする
  ;; - 再帰的な保存ループを防ぐため、バッファローカルな実行中フラグを用いる
  ;; - 名前付きフック関数 `orgx-auto-tangle--install-hook' を介して、
  ;;   `orgx-auto-tangle--maybe' を `after-save-hook' にローカル登録する
  ;; - org / ob-tangle をトップレベルで require しない（Org の遅延ロードを保つ）
  ;;; Code:

  ;; org と ob-tangle はトップレベルで require しない（Org を遅延ロードするため）。
  ;; 本モジュールのフックは org-mode-hook 経由でしか登録されないので、
  ;; orgx-auto-tangle--maybe が走る時点で org は必ずロード済みである。
  ;; ob-tangle だけを、実際に tangle する直前に require する。
  (declare-function org-babel-tangle "ob-tangle" (&optional arg target-file lang-re))

  (declare-function orgx-core-source-org-p "orgx-core" (&optional file))
  (defvar-local orgx-auto-tangle--in-progress nil)

  (defun orgx-auto-tangle--eligible-p ()
    "Return non-nil when the current buffer is one of the Org sources."
    (and (derived-mode-p 'org-mode)
         (require 'orgx-core nil t)
         (orgx-core-source-org-p)))

  ;; 特殊変数であることをコンパイラへ伝える（Org を読み込まずにコンパイルされても、
  ;; 下の let が動的束縛として扱われるようにするため）。宣言が無いと、この let は字句的束縛になり、確認の抑止が効かない。
  (defvar org-confirm-babel-evaluate)

  (defun orgx-auto-tangle--maybe ()
    (when (and (orgx-auto-tangle--eligible-p)
               (not orgx-auto-tangle--in-progress))
      (require 'ob-tangle)
      (let ((orgx-auto-tangle--in-progress t)
            (org-confirm-babel-evaluate nil)
            (inhibit-modification-hooks t))
        (condition-case err
            (org-babel-tangle)
          (error
           (message "[orgx-auto-tangle] error: %s" (error-message-string err)))))))

  ;; 無名 lambda を名前付き defun に抽出した。
  ;; 理由: org-mode-hook 上の無名 lambda は remove-hook で除去できず、
  ;; describe-function で検査もできず、eval-buffer をまたいだ重複登録に対しても
  ;; 安全でない。
  (defun orgx-auto-tangle--install-hook ()
    "Install the auto-tangle after-save hook buffer-locally."
    (add-hook 'after-save-hook #'orgx-auto-tangle--maybe nil 'local))

  (add-hook 'org-mode-hook #'orgx-auto-tangle--install-hook)

  ;; Emacs 31: timeout.el によるスロットリング
  ;; ---------------------------------------------------------------------------
  ;; 保存した Org 文書の tangle は、保存が連続すると（auto-save-visited-mode に加えて
  ;; 明示的な C-x C-s）2 回走る。
  ;; `timeout-throttle' は advice を設置し、一定間隔内の繰り返し呼び出しを 1 回に
  ;; まとめる。上記の実行中フラグは維持する。スロットリングは頻度を制限するだけで、
  ;; 非再入性を保証するものではないためである。
  (when (and (require 'timeout nil t) (fboundp 'timeout-throttle))
    (timeout-throttle #'orgx-auto-tangle--maybe 2.0))

  (provide 'orgx-auto-tangle)

Hugo ブログ（ox-hugo）に関わる設定を 1 つにまとめたモジュールです。

以前は、ox-hugo の導入が =orgx-export.el= に、ブログ記事の capture テンプレート "b" が =orgx-core.el= に、
そのテンプレートの上書きが =personal/personal-hugo.el= に分かれていました。 =orgx-export.el= は
326 行のうち約 45 行だけが Hugo で、残りは Markdown、Mermaid、dot、LaTeX の話題でした。
=orgx-core.el= と =personal-hugo.el= では、同じテンプレートの本文を二重に持っていました。

このモジュールが持つのは次の 2 つです。

- ox-hugo の導入と、ディスパッチメニュー・エクスポート時の 2 つのアドバイス（ =orgx-export.el= から移した）
- ブログ記事の capture テンプレート "b"（タグを入力するかは =orgx-hugo-capture-prompt-tags-p= で切り替え）

タグを capture の確定前に別のコマンドで設定する運用の個人設定は、このオプションを =nil= にするだけです。
テンプレートを丸ごと上書きする必要はありません。 =orgx-core.el= の =org= の設定のあとに追加するため、
=my:modules= では =orgx-core= より後（ =orgx-auto-hugo= の直前）に置いています。

  ;;

  ;; ロードしない。ox は org を require するため、それが Org を起動時に読み込む経路の

  ;; :custom による先行登録は ox-hugo のロード時に反映される。ox-hugo は ox を、ox は org を
  ;; require するため、:require t を付けると「Org の遅延ロード」の効果がここから漏れて

  ;; ブログ記事の capture かどうかの判定は、ここの 1 か所にまとめる。
  ;; 判定の基準は「保存先のバッファが `my:f:capture-blog-file' を訪問しているか」で、

  (defun orgx-auto-hugo-preview-capture ()
    "Render the entry currently being captured through ox-hugo and show it in
  `markdown-preview-mode' as a one-shot browser preview. Does not finalize or
  abort the capture, and never writes into the real Hugo content tree.

  Uses `org-export-as' with the `hugo' backend and SUBTREEP non-nil to export
  only the current entry to a string, entirely in memory — this deliberately
  avoids `org-hugo-export-wim-to-md' and `org-hugo-base-dir', since the
  capture target file may already set #+HUGO_BASE_DIR, which would take
  precedence over any dynamic let-binding and could export straight into the
  real content directory. The resulting string is written to a per-session
  scratch file (see `orgx-auto-hugo-preview--scratch-file') and previewed
  from there.

  Intended to be bound in `org-capture-mode-map'; signals a `user-error'
  outside a capture buffer.

  `markdown-preview-mode' reads the preview buffer's *window* when it pushes
  an update to the browser (`markdown-preview--send-preview-to', used for
  scroll-sync). Called on a buffer that was never actually displayed — as
  happens with plain `find-file-noselect' — it errors with \"Wrong type
  argument: number-or-marker-p, nil\" instead of pushing the update (the
  scratch file itself still gets written correctly; only the push fails).
  To avoid that, the scratch buffer is popped into a real window for the
  duration of the update, then focus is restored to the capture buffer so
  typing is not interrupted."
    (interactive)
    (unless (bound-and-true-p org-capture-mode)
      (user-error "orgx-auto-hugo-preview-capture: not in an org-capture buffer"))
    (require 'ox-hugo)
    (require 'markdown-preview-mode)
    (let ((md-string
           (condition-case err
               (org-export-as 'hugo t nil nil nil)
             (error
              (message "[orgx-auto-hugo] preview export failed: %s"
                       (error-message-string err))
              nil))))
      (when md-string
        (let* ((capture-buf (current-buffer))
               (file (orgx-auto-hugo-preview--scratch-file))
               (buf (find-file-noselect file)))
          (save-selected-window
            (pop-to-buffer buf)
            (erase-buffer)
            (insert md-string)
            (save-buffer)
            ;; The mode's own initial "send to browser" can still throw the
            ;; error above in edge cases (e.g. the websocket client has not
            ;; connected yet); by that point the server has already started
            ;; and the buffer-local mode is already on, so it is safe to
            ;; swallow and let the next save (or a manual browser reload of
            ;; the printed URL) pick up the content.
            (ignore-errors
              (unless (bound-and-true-p markdown-preview-mode)
                (markdown-preview-mode 1))))
          (when (buffer-live-p capture-buf)
            (pop-to-buffer capture-buf))
          (message "[orgx-auto-hugo] preview updated.")))))

** orgx-roam-ui.el — simple-httpd の local-repo 衝突

本モジュールは長らく =(require 'simple-httpd)= の失敗に悩まされ、 =(require 'web-server)= と
=(provide 'simple-httpd)= のシムで回避していました。この回避策は誤診に基づくもので、2026-07 に
根本原因を特定して撤去しました。

*誤診の内容*

「MELPA の =simple-httpd= が provide する feature シンボルは =web-server= である」と結論して
いましたが、これは事実ではありません。実際には次のとおりです。

| 事実 | 内容 |
|------+------|
| MELPA recipe =simple-httpd= | =(:repo "skeeto/emacs-web-server" :fetcher github)= |
| MELPA recipe =web-server=   | =(:repo "eschulte/emacs-web-server")= — 別作者・別パッケージ・ *同じリポジトリ名* |
| =skeeto/emacs-web-server= の中身 | =simple-httpd.el= と =simple-httpd-test.el= の 2 つのみ。 =web-server.el= は存在しない |
| =simple-httpd.el= | 末尾に =(provide 'simple-httpd)= 。 =defservlet= / =defservlet*= を =httpd-servlet= / =httpd-servlet*= への =defalias= として定義する |

=skeeto/emacs-http-server= は =skeeto/emacs-web-server= にリネームされています。GitHub が
リダイレクトするので旧名でも取得できてしまい、これが混乱の入口になっていました。

*真の原因: straight の local-repo 名の衝突*

straight の =:local-repo= は、既定でリポジトリ名になります。そのため、次の衝突が起きます。

#+begin_example
simple-httpd → skeeto/emacs-web-server   → local-repo "emacs-web-server"
web-server   → eschulte/emacs-web-server → local-repo "emacs-web-server"   ← 同一
#+end_example

先に clone された側がディレクトリを占有し、もう一方はそれを使い回します。eschulte 版が居座った
結果、 =build-XX.Y/simple-httpd/= に =web-server.el= が並ぶ、という観測になっていました。

そこへ =(provide 'simple-httpd)= のシムを足したので、 =require= は通るのに =defservlet*= は定義
されない状態が固定化されました。 =defservlet*= は =simple-httpd.el= にしか無いためです。
=org-roam-ui.el= はトップレベルで =(defservlet* node/:id ...)= を呼ぶので、ロード時に
=void-function defservlet*= で落ちます。

*対処*

=simple-httpd= に明示的な =:local-repo= を与えて分離します。シムと =(require 'web-server)= は
撤去しました。upstream の =org-roam-ui= には欠陥が無く、フォークは不要です。

すでに汚染された環境では、ディレクトリの掃除が必要です。

#+begin_example
rm -rf ~/.emacs.d/.cache/straight/repos/emacs-web-server
rm -rf ~/.emacs.d/.cache/straight/build-*/simple-httpd
#+end_example

*トップレベル require の全廃*

依存の導入（ =straight-use-package= ）だけをトップレベルで行い、 =require= はしません。
=org-roam-ui-open= には =;;;###autoload= が付いていて、呼び出し時に自動でロードされます。
これで、本モジュールが =org-roam-ui= → =org-roam= → =org= の連鎖を起動時に引き込むことは
なくなり、Org の遅延ロードが保たれます。

#+begin_src emacs-lisp
  ;;; orgx/orgx-roam-ui.el --- org-roam-ui グラフビジュアライザ -*- lexical-binding: t; -*-

  ;; Author: YAMASHITA, Takao <tjy1965@gmail.com>
  ;; Category: org
  ;;
  ;;; Commentary:
  ;;
  ;; org-roam ノードの対話的グラフビジュアライザ。ローカルの HTTP + WebSocket サーバ経由で
  ;; 提供され、ブラウザで閲覧する。
  ;;
  ;; 本ファイルの責務:
  ;; - simple-httpd、websocket、org-roam-ui をトップレベルで導入する
  ;;   （straight-use-package のみ。require はしない）
  ;; - simple-httpd には明示的な :local-repo を与え、web-server との local-repo 名の
  ;;   衝突を防ぐ（下記の注記を参照）
  ;; - setopt による設定は with-eval-after-load 'org-roam に遅延する
  ;; - org-roam がロードされた後、一度きりのアイドルタイマー（8 秒）で UI を開く
  ;;   （orgx-extensions は autosync を自動起動しないので、DB の同期は待たない）
  ;; - セッション中に org-roam が強制再読込された場合の重複タイマー登録を防ぐため、
  ;;   defvar のガードを用いる
  ;;
  ;; require をトップレベルで行わない理由:
  ;;   org-roam-ui-open には ;;;###autoload が付いている。straight-use-package が
  ;;   ビルドと autoload 生成を済ませるため、呼び出し時に自動でロードされる。
  ;;   ここで require すると org-roam-ui -> org-roam -> org の連鎖を起動時に引き込み、
  ;;   Org の遅延ロードが台無しになる。
  ;;
  ;; ロード経路: personal/user.el が my:modules-extra に追加する。
  ;; 本モジュールは正規の my:modules リストに現れてはならない。
  ;;
  ;;; Code:

  ;; simple-httpd の local-repo 衝突を回避する。
  ;;
  ;; MELPA のレシピは次のようになっている。
  ;;   simple-httpd → skeeto/emacs-web-server
  ;;   web-server   → eschulte/emacs-web-server   (別作者・別パッケージ・同名リポジトリ)
  ;;
  ;; straight の :local-repo は既定でリポジトリ名になるため、両者とも
  ;; "emacs-web-server" を要求して衝突する。先に clone された側がディレクトリを占有し、
  ;; もう一方はその中身を使い回してしまう。eschulte 版が居座ると
  ;; build-XX.Y/simple-httpd/ に web-server.el が並び、(require 'simple-httpd) は失敗する。
  ;;
  ;; defservlet* は simple-httpd.el にしか存在しない（httpd-servlet* への defalias）。
  ;; org-roam-ui.el はトップレベルで (defservlet* node/:id ...) を呼ぶため、誤った
  ;; simple-httpd をロードすると void-function defservlet* で落ちる。
  ;;
  ;; 明示的な :local-repo と :files でこれを分離する。
  (condition-case err
      (straight-use-package
       '(simple-httpd :type git :host github
                      :repo "skeeto/emacs-web-server"
                      :local-repo "simple-httpd"
                      :files ("simple-httpd.el")))
    (error
     (message "[orgx-roam-ui] simple-httpd install failed: %s"
              (error-message-string err))))

  (condition-case err
      (straight-use-package 'websocket)
    (error
     (message "[orgx-roam-ui] websocket install failed: %s"
              (error-message-string err))))

  (condition-case err
      (straight-use-package
       '(org-roam-ui
         :type git
         :host github
         :repo "org-roam/org-roam-ui"
         :branch "main"
         :files ("*.el" "out")))
    (error
     (message "[orgx-roam-ui] org-roam-ui install failed: %s"
              (error-message-string err))))

  (defvar orgx-roam-ui--timer-registered nil
    "Non-nil when the idle timer for `orgx-roam-ui--open-deferred' has been
  registered.  Guards against duplicate timer registration when org-roam is
  force-reloaded during the same session.")

  (defun orgx-roam-ui--open-deferred ()
    "Open org-roam-ui once, after org-roam has loaded.
  Scheduled via an 8-second idle timer.  Note that orgx-extensions no
  longer starts the org-roam autosync by itself.
  `org-roam-ui-open' is autoloaded, so org-roam-ui is loaded here on demand."
    (condition-case err
        (when (fboundp 'org-roam-ui-open)
          (org-roam-ui-open)
          (message "[org-roam-ui] graph opened"))
      (error
       (message "[org-roam-ui] open failed: %s"
                (error-message-string err)))))

  (with-eval-after-load 'org-roam
    ;; org-roam-ui-* は org-roam-ui.el の defcustom である。setopt は未ロードの
    ;; defcustom に対しても値を設定でき、後から defcustom が評価されても束縛済みの値は
    ;; 上書きされない。
    (setopt org-roam-ui-sync-theme     t
            org-roam-ui-follow         t
            org-roam-ui-update-on-save t)
    (unless orgx-roam-ui--timer-registered
      (run-with-idle-timer 8 nil #'orgx-roam-ui--open-deferred)
      (setq orgx-roam-ui--timer-registered t)))

  (provide 'orgx-roam-ui)
  ;;; orgx/orgx-roam-ui.el ends here
#+end_src
