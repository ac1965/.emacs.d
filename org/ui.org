  ;;; ui/ui-font.el --- フォント設定 -*- lexical-binding: t; -*-
  ;;
  ;; Copyright (c) 2021-2026
  ;; Author: YAMASHITA, Takao
  ;; License: GNU GPL v3 or later
  ;;
  ;; Category: ui
  ;;
  ;;; Commentary:
  ;;
  ;; フォント設定: 等幅、可変ピッチ、絵文字、リガチャ。
  ;;
  ;; 本ファイルの責務:
  ;; - `ui-font-*' カスタマイズ変数を定義する（フォントファミリ、サイズ、絵文字）
  ;; - `ui-font--system-default' と `ui-font--system-emoji' により
  ;;   OS を考慮したフォールバックを提供する
  ;; - `ui-font-apply' で全フェイスを適用し、`after-init-hook' と
  ;;   `after-make-frame-functions' にフックする（daemon 対応）
  ;; - グラフィカルフレームで `global-ligature-mode' を有効化する
  ;;   （初回のグラフィカルフレーム作成時に一度だけ。daemon 対応）
  ;;; Code:

  (eval-when-compile (require 'leaf))

  (defgroup ui-font nil "Font configuration for UI." :group 'ui)

  (defcustom ui-font-default nil
    "Default monospace font family. If nil, OS-dependent fallback."
    :type '(choice (const :tag "Auto" nil) (string :tag "Font family"))
    :group 'ui-font)

  (defcustom ui-font-variable-pitch nil
    "Variable-pitch font family."
    :type '(choice (const :tag "Auto" nil) (string :tag "Font family"))
    :group 'ui-font)

  (defcustom ui-font-emoji nil
    "Emoji font family."
    :type '(choice (const :tag "Auto" nil) (string :tag "Font family"))
    :group 'ui-font)

  (defcustom ui-font-size 18
    "Default font size in points."
    :type 'integer :group 'ui-font)

  (defun ui-font--system-default ()
    "Return the default monospace font family for the current OS."
    (cond ((eq system-type 'darwin)     "Menlo")
          ((eq system-type 'gnu/linux)  "Monospace")
          ((eq system-type 'windows-nt) "Consolas")
          (t "Monospace")))

  (defun ui-font--system-emoji ()
    "Return the default emoji font family for the current OS."
    (cond ((eq system-type 'darwin)     "Apple Color Emoji")
          ((eq system-type 'gnu/linux)  "Noto Color Emoji")
          ((eq system-type 'windows-nt) "Segoe UI Emoji")
          (t "Noto Color Emoji")))

  (defun ui-font-apply ()
    "Apply configured fonts to the current graphical frame."
    (when (display-graphic-p)
      (set-face-attribute 'default nil
                          :family (or ui-font-default (ui-font--system-default))
                          :height (* 10 ui-font-size))
      (set-face-attribute 'variable-pitch nil
                          :family (or ui-font-variable-pitch
                                      ui-font-default
                                      (ui-font--system-default)))
      (set-fontset-font t 'emoji
                        (font-spec :family (or ui-font-emoji (ui-font--system-emoji))))
      (ui-font-describe)))

  ;;;###autoload
  (defun ui-font-show-current ()
    "Display the currently active default font."
    (interactive)
    (let ((family (face-attribute 'default :family))
          (height (face-attribute 'default :height)))
      (message "Current font: %s, %.1f pt" family (/ height 10.0))))

  ;;;###autoload
  (defun ui-font-describe ()
    "Show detailed information about the current font configuration."
    (interactive)
    (unless (display-graphic-p)
      (user-error "Fonts are only meaningful in graphical frames"))
    (let* ((default-family  (face-attribute 'default :family))
           (default-height  (face-attribute 'default :height))
           (variable-family (face-attribute 'variable-pitch :family))
           ;; 絵文字のフォントファミリは、ui-font-apply が set-fontset-font に渡した
           ;; defcustom／フォールバックの値から読む。(font-get (font-spec :script 'emoji) :family) で
           ;; 探す方法は使えない。登録済みの fontset
           ;; エントリではなくフォントの利用可否を問い合わせるため、macOS では nil を
           ;; 返し「unknown」と報告されてしまう。
           (emoji-family    (or ui-font-emoji (ui-font--system-emoji))))
      (message "Fonts applied:\n  Default        : %s (%.1f pt)\n  Variable-pitch : %s\n  Emoji          : %s"
               default-family (/ default-height 10.0)
               variable-family emoji-family)))

  ;; 無名 lambda を名前付き defun に抽出した。
  ;; 理由: デーモン分岐が after-make-frame-functions に無名関数を登録していたため、
  ;; 除去も検査もできなかった。
  (defun ui-font--apply-to-frame (frame)
    "Apply font configuration to FRAME (for daemon mode)."
    (with-selected-frame frame (ui-font-apply)))

  (if (daemonp)
      (add-hook 'after-make-frame-functions #'ui-font--apply-to-frame)
    (add-hook 'after-init-hook #'ui-font-apply))

  ;; leaf の :when は「モジュールのロード時」に一度だけ評価される。通常起動では
  ;; その時点で既にグラフィカルフレームがあるため (display-graphic-p) は機能するが、
  ;; --daemon 起動時はモジュールのロード自体がフレームの無い状態で行われるため、
  ;; :when (display-graphic-p) は常に nil と評価され、ligature が永久にロードされない
  ;; という daemon 対応バグになっていた。フォント適用（上の ui-font--apply-to-frame）と
  ;; 同じ「初回のグラフィカルフレームで一度だけ有効化する」パターンに揃える。
  (leaf ligature
    :straight (ligature :type git :host github :repo "mickeynp/ligature.el")
    :require nil)

  (defvar ui-font--ligature-activated-p nil
    "Non-nil once ligature has been configured and enabled.")

  (defun ui-font--activate-ligature ()
    "ligature を require し、prog-mode 用の定義とグローバル有効化を行う。
  `ui-font--maybe-activate-ligature' から一度だけ呼ばれる。"
    (require 'ligature)
    ;; prog-mode 用にリガチャを登録してから、グローバルに有効化する。
    ;; global-ligature-mode はグローバルなマイナーモードであるため、prog-mode-hook に
    ;; 追加してはならない — 繰り返し有効化するのは冗長であり、デバッグを混乱させる。
    (ligature-set-ligatures 'prog-mode '("->" "=>" "::" "===" "!=" "&&" "||"))
    (global-ligature-mode 1))

  (defun ui-font--maybe-activate-ligature (&optional frame)
    "FRAME がグラフィカルなら、初回に限り ligature を有効化する。
  `after-make-frame-functions' （daemon）と `after-init-hook' （通常起動）の
  両方から呼ばれる。通常起動の =after-init-hook= には引数が渡らないため、
  FRAME は省略可能にし、その場合は `selected-frame' を見る。"
    (when (and (not ui-font--ligature-activated-p)
               (display-graphic-p (or frame (selected-frame))))
      (setq ui-font--ligature-activated-p t)
      (ui-font--activate-ligature)))

  (if (daemonp)
      (add-hook 'after-make-frame-functions #'ui-font--maybe-activate-ligature)
    (add-hook 'after-init-hook #'ui-font--maybe-activate-ligature))

  (provide 'ui-font)
  ;;; ui-font.el ends here
#+end_src

  ;; Copyright (c) 2021-2026
  ;; Author: YAMASHITA, Takao

    "Critical error color."
    :type 'color

    "Informational color."
    :type 'color

  ;;;; ---------------------------------------------------------------------------

  ;;;; ---------------------------------------------------------------------------

  (defcustom ui-nano-palette-color-selection "#dbeafe"

    :type 'color

  ;;;; ---------------------------------------------------------------------------
  ;;;; ユーティリティ
  ;;;; ---------------------------------------------------------------------------

  (defvar ui-nano-palette--touched-faces nil
    "`ui-nano-palette--safe-face' が属性を設定したフェイスの一覧。
  dark へ切り替えるとき、テーマが定義していない（light の固定色が残る）フェイスを
  元の定義へ戻すために使う。")

  (defun ui-nano-palette--safe-face (face &rest args)
    "Safely apply FACE attributes."
    (when (facep face)
      (cl-pushnew face ui-nano-palette--touched-faces)
      (apply #'set-face-attribute face nil args)))

  ;;;; ---------------------------------------------------------------------------
  ;;;; フェイスの適用
  ;;;; ---------------------------------------------------------------------------

  ;; このパレットは明るい背景を前提にした固定色である。`nano-dark' に切り替えた後も
  ;; `default' や font-lock を明るい配色で上書きしてしまい、light と dark の見た目が
  ;; 同じになっていた。dark のときはパレットを適用せず、nano-theme 側のフェイスに任せる。
  ;; dark では、light のとき set-face-attribute で焼いた固定色を戻す。テーマが
  ;; 定義するフェイスは enable-theme が再計算するが、dirvish-hl-line のように
  ;; テーマが触れないフェイスには明るい背景色が残り、白い文字が読めなくなる。
  ;; `default' はフォント設定を失わないよう対象から除く。
  ;; テーマ（nano-dark）が定義しない dired / dirvish のフェイスには、dark 用の色を
  ;; nano-theme 自身のパレット（`nano-dark-*'、Nord 系）から与える。独自の dark 色は
  ;; 作らない。`nano-dark-*' が未定義のときは何も設定しない（元の定義のまま）。
  (defun ui-nano-palette--dark-color (symbol)
    "`nano-theme' の dark パレット SYMBOL の色を返す。未定義なら nil。"
    (let ((value (and (boundp symbol) (symbol-value symbol))))
      (and (stringp value) value)))

  ;; フェイスの一覧。(FACE :属性 値 ...) の並びで、値が色の役割（ROLE）の名前なら、
  ;; 適用時にパレットの色へ解決する。(box ROLE) は 1px の枠線の指定になる。
  ;; ROLE 以外の値（bold、semi-bold、italic、t など）はそのまま渡す。
  (defconst ui-nano-palette--roles
    '(background foreground subtle strong muted salient popout success warning
                 critical info selection cursor)
    "`ui-nano-palette-color-ROLE' の ROLE の一覧（light 用）。")

  (defconst ui-nano-palette--dark-roles
    '(salient faded critical strong popout subtle)
    "`nano-dark-ROLE' の ROLE の一覧（dark の追加フェイス用）。")

  (defconst ui-nano-palette--light-faces
    '(;; 既定フェイス
      (default :background background :foreground foreground)
      ;; リージョン
      (region :background selection)
      ;; 現在行のハイライト（global-hl-line-mode）。
      ;; dirvish-hl-line と同じ色を使い、バッファの種類を問わず見た目を揃える。
      ;; この設定が抜けていたため、global-hl-line-mode を有効化しても
      ;; 通常の編集バッファでは現在行の背景色が付かなかった。
      (hl-line :background subtle)
      ;; 太字 / 斜体
      (bold :foreground strong :weight bold)
      (italic :slant italic)
      ;; フォントロック
      (font-lock-comment-face :foreground muted)
      (font-lock-keyword-face :foreground salient :weight semi-bold)
      (font-lock-string-face :foreground popout)
      (font-lock-function-name-face :foreground info)
      (font-lock-type-face :foreground salient)
      (font-lock-variable-name-face :foreground strong)
      ;; メッセージ
      (error :foreground critical :weight bold)
      (warning :foreground warning :weight bold)
      (success :foreground success)
      (link :foreground salient :underline t)
      (minibuffer-prompt :foreground salient :weight semi-bold)
      ;; モードライン
      (mode-line :background subtle :foreground strong :box (box subtle))
      (mode-line-inactive :background background :foreground muted
                          :box (box background))
      ;; dired / dirvish
      (dired-directory :foreground salient :weight bold)
      (dired-symlink :foreground info)
      (dired-ignored :foreground muted)
      (dired-flagged :foreground critical :weight bold)
      (dired-marked :foreground salient :weight bold)
      (dired-header :foreground strong :weight bold)
      (dired-warning :foreground warning :weight bold)
      (dirvish-hl-line :background subtle)
      (dirvish-git-commit-message-face :foreground muted)
      (dirvish-header-line-path-face :foreground strong :weight bold)
      (dirvish-subtree-state-face :foreground salient))
    "light のパレットで設定するフェイスの一覧。")

  (defconst ui-nano-palette--dark-faces
    '((dired-directory :foreground salient :weight bold)
      (dired-symlink :foreground salient)
      (dired-ignored :foreground faded)
      (dired-flagged :foreground critical :weight bold)
      (dired-marked :foreground salient :weight bold)
      (dired-header :foreground strong :weight bold)
      (dired-warning :foreground popout :weight bold)
      (dirvish-hl-line :background subtle)
      (dirvish-git-commit-message-face :foreground faded)
      (dirvish-header-line-path-face :foreground strong :weight bold)
      (dirvish-subtree-state-face :foreground salient))
    "dark で、テーマが定義しない dired / dirvish のフェイスに与える色の一覧。")

  (defun ui-nano-palette--resolve (value roles color-fn)
    "フェイスの一覧の属性値 VALUE を、`set-face-attribute' に渡す値へ解決する。
  VALUE が ROLES の要素なら COLOR-FN で色にし、(box ROLE) は枠線の指定にする。
  それ以外は VALUE をそのまま返す。"
    (cond ((and (symbolp value) (memq value roles))
           (funcall color-fn value))
          ((eq (car-safe value) 'box)
           (list :line-width 1 :color (funcall color-fn (cadr value))))
          (t value)))

  (defun ui-nano-palette--apply-specs (specs roles color-fn)
    "SPECS（フェイスの一覧）を `ui-nano-palette--safe-face' で適用する。
  ROLES と COLOR-FN は `ui-nano-palette--resolve' に渡す。"
    (dolist (spec specs)
      (let ((attrs (cdr spec)) resolved)
        (while attrs
          (push (pop attrs) resolved)
          (push (ui-nano-palette--resolve (pop attrs) roles color-fn) resolved))
        (apply #'ui-nano-palette--safe-face (car spec) (nreverse resolved)))))

  (defun ui-nano-palette--dark-role-color (role)
    "dark パレットの ROLE の色を返す。未定義なら nil。"
    (ui-nano-palette--dark-color (intern (format "nano-dark-%s" role))))

  (defun ui-nano-palette--apply-dark-extras ()
    "dired / dirvish のフェイスに `nano-dark-*' の色を与える。
  必要な色がひとつでも未定義なら、何も設定しない（元の定義のまま）。"
    (when (seq-every-p #'ui-nano-palette--dark-role-color
                       ui-nano-palette--dark-roles)
      (ui-nano-palette--apply-specs ui-nano-palette--dark-faces
                                    ui-nano-palette--dark-roles
                                    #'ui-nano-palette--dark-role-color)))

  (defun ui-nano-palette-apply-faces ()
    "Apply Nano palette; in dark, restore touched faces and apply dark extras."
    (if (eq frame-background-mode 'dark)
        (progn
          (dolist (face ui-nano-palette--touched-faces)
            (unless (eq face 'default)
              (face-spec-recalc face nil)))
          ;; light パレットが `set-face-attribute' で default に焼き込んだ背景色は
          ;; face-spec-recalc では消えない。通常起動では light の適用後に dark へ
          ;; 切り替わるため、dark の色を明示して上書きする。
          (let ((bg (ui-nano-palette--dark-color 'nano-dark-background))
                (fg (ui-nano-palette--dark-color 'nano-dark-foreground)))
            (when (and bg fg)
              (ui-nano-palette--safe-face 'default :background bg :foreground fg)))
          (ui-nano-palette--apply-dark-extras))
      (ui-nano-palette--apply-palette)))

  (defun ui-nano-palette--light-color (role)
    "light パレットの ROLE の色を返す。"
    (symbol-value (intern (format "ui-nano-palette-color-%s" role))))

  (defun ui-nano-palette--apply-palette ()
    "Apply the (light) Nano palette."
    ;; カーソル
    (setq cursor-type 'bar)
    (set-cursor-color ui-nano-palette-color-cursor)
    (ui-nano-palette--apply-specs ui-nano-palette--light-faces
                                  ui-nano-palette--roles
                                  #'ui-nano-palette--light-color))

  (defun ui-nano-palette--reapply-after-theme (&rest _)
    "Reapply Nano palette after theme load."
    (ui-nano-palette-apply-faces))

  (advice-add 'load-theme :after #'ui-nano-palette--reapply-after-theme)

  ;; dirvish は dired-noselect 初回呼び出し時まで require されない
  ;; （core/core-dired.el の `core-dired--ensure' 参照）。
  ;; そのため起動時の `ui-nano-palette-apply-faces' 呼び出し時点では
  ;; `dirvish-hl-line' 等がまだ defface されておらず、`ui-nano-palette--safe-face'
  ;; の facep チェックに弾かれて黙ってスキップされていた。dirvish は
  ;; カーソルを非表示にしてこのフェイスのオーバーレイだけでカレント行を
  ;; 示す設計のため、パレット未適用のままだと現在行が全く見えなくなる。
  ;; dirvish が実際に require されたタイミングで一度だけ再適用する。
  (with-eval-after-load 'dirvish
    (ui-nano-palette-apply-faces))

  (provide 'ui-nano-palette)

  ;;; ui/ui-nano-palette.el ends here
#+end_src

    `%' is doubled because the header line interprets it as a format spec.  The
    text uses the secondary face, like the rest of the right segment, so its

    (let ((str (and ui-nano-modeline-extra-function

    "Return a propertized \"HH:MM  CODING  IME\" string for the right segment.

    Evaluated on every redisplay inside `nano-modeline--make'.  Time and coding

  (defun ui-nano-modeline--battery ()
    "Return \"AC 87%  \" or \"BATT 42%  \" for the nano-modeline right segment.

    The text comes from `ui-nano-modeline--battery-data' (kept current by
    `display-battery-mode').  On battery it uses the
    `ui-nano-modeline-battery-low' and `ui-nano-modeline-battery-critical'
    faces below `battery-load-low' and `battery-load-critical'.  Returns an
    empty string when there is no battery information, for example before the
    first update or on a machine without a battery."
    (let* ((data ui-nano-modeline--battery-data)
           (source (cdr (assq ?L data)))
           (load (cdr (assq ?p data)))
           (percent (and (stringp load) (string-match-p "\\`[0-9]+\\'" load)
                         (string-to-number load)))

           (base (when (fboundp 'nano-modeline-face)
                   (nano-modeline-face 'secondary))))
      (if (not (and percent (member source '("AC" "Battery"))))
          ""
        (propertize
         ;; モードラインは :eval の結果の % を解釈するため、% を %% に重ねて書く。
         (format "%s %d%%%%  " (if on-ac "AC" "BATT") percent)
         'face (cond ((and (not on-ac) (< percent battery-load-critical))
                      (list 'ui-nano-modeline-battery-critical base))
                     ((and (not on-ac) (< percent battery-load-low))
                      (list 'ui-nano-modeline-battery-low base))
                     (t base))))))

    '((t :inherit ui-nano-modeline-status-icon :background "#FFAB40" :foreground "#3E2723"))
    "Face of the status icon for a modified buffer (stronger amber)."
    :group 'ui-nano-modeline)

    '((t :inherit ui-nano-modeline-status-icon :background "#FFE9B8"))
    "Face of the status icon for a read-only buffer (paler amber)."
    :group 'ui-nano-modeline)

  (defun ui-nano-modeline--type-icon ()
    "Return the file type icon of the current buffer for the nano-modeline.

    The icon comes from `nerd-icons-icon-for-buffer' (by major mode), so it keeps
    its own colour and icon font.  Only the background is added, by inheriting
    `nano-modeline-active', so that it sits on the same band as the rest of the
    line.  Returns an empty string when there is no icon."
    (let* ((icon  (nerd-icons-icon-for-buffer))
           (props (and (stringp icon) (> (length icon) 0)
                       (get-text-property 0 'face icon))))
      (if (not (and (consp props) (plist-get props :family)))
          ""
        (concat
         (propertize " " 'face 'nano-modeline-active)
         (propertize (substring-no-properties icon)
                     'face (list :family (plist-get props :family)
                                 :inherit (delq nil (list (plist-get props :inherit)
                                                          'nano-modeline-active))))))))

  (defun ui-nano-modeline--buffer-status ()
    "Return the buffer status segment, shown as an icon when possible.

    Read-only buffers get a lock, modified ones a pencil and the rest a file
    icon, on amber backgrounds that vary by state (`ui-nano-modeline-status-icon'
    and its -modified / -read-only variants).  The
    vertical padding is `nano-modeline-padding'.  Falls back to
    `nano-modeline-buffer-status' (the text \"RO\" / \"**\" / \"RW\") on a
    non-graphical display or when `nerd-icons' is unavailable.

    The whole segment gets one face, so the icon font is set here as a separate
    face ahead of it, and the padding spaces carry the face too.  The file type
    icon of the buffer (`ui-nano-modeline--type-icon') follows the status icon."
    (if (not (and (display-graphic-p) (require 'nerd-icons nil t)))
        (nano-modeline-buffer-status)
      (let* ((name  (cond (buffer-read-only      "nf-fa-lock")
                          ((buffer-modified-p)   "nf-fa-pencil")
                          (t                     "nf-fa-file_o")))
             (glyph (substring-no-properties (nerd-icons-faicon name)))
             (face  (cond (buffer-read-only      'ui-nano-modeline-status-icon-read-only)
                          ((buffer-modified-p)   'ui-nano-modeline-status-icon-modified)
                          (t                     'ui-nano-modeline-status-icon)))
             (gap   (propertize " " 'display '(space :width 0.4)))
             (top   (propertize " " 'display `(raise ,(car nano-modeline-padding))))
             (bot   (propertize " " 'display `(raise ,(- (cdr nano-modeline-padding))))))
        ;; top / bot（raise を指定しただけの空白）にも face が必要。
        ;; 付けないと、アイコンの周りだけ地の色が見えて継ぎ目になる。
        (concat
         (propertize (concat top gap) 'face face)
         (propertize glyph 'face (list '(:family "Symbols Nerd Font Mono") face))
         (propertize (concat gap bot) 'face face)
         (ui-nano-modeline--type-icon)))))

  (defun ui-nano-modeline--line (&optional default)
    "Install the nano-modeline layout used for every buffer.

    DEFAULT is passed through to `nano-modeline-header':
    t means global (setq-default), nil means buffer-local (setq-local).
    The left segment shows the status icon, name, git info, file size and line
    count; the right segment shows the extra item, battery, time, encoding and
    cursor position."

               (nano-modeline-git-info) " "
               (nano-modeline-file-size) " "

               (ui-nano-modeline--battery)
               (ui-nano-modeline--time-encoding) " "
               (nano-modeline-cursor-position)
               (nano-modeline-window-dedicated))
             default))

  (defun ui-nano-modeline--dark-foreground ()
    "dark の前景色。nano-theme の `nano-dark-foreground' が使えなければ既定の色を返す。"
    (let ((value (and (boundp 'nano-dark-foreground) nano-dark-foreground)))
      (if (and (stringp value) (not (string-empty-p value))) value "#ECEFF4")))

  ;; nano-modeline の `nano-modeline-active' / `-inactive' は defface のロード時に
  ;; その時点の default / header-line の色を焼き込む。nano-theme はこの 2 つの
  ;; 定義をコメントアウトしている（`(:inherit nano-subtle)' 相当を想定）ため、
  ;; light で読み込まれた値が dark に切り替えた後も残り、明るい背景に dark の
  ;; 文字色（白）が載って判読できなくなる。テーマ追従の `nano-subtle' を継承させる。
  (defun ui-nano-modeline-sync-faces ()
    "nano-modeline の基底フェイスを現在のテーマ（light / dark）に追従させる。"
    (when (and (facep 'nano-modeline-active) (facep 'nano-subtle))
      (dolist (face '(nano-modeline-active nano-modeline-inactive))
        (set-face-attribute face nil
                            :inherit 'nano-subtle
                            :foreground 'unspecified
                            :background 'unspecified
                            :box nil)))
    ;; `nano-modeline--empty-face' は header-line の土台（face-remap 先）で、
    ;; defface のロード時の default 前景色（light では Black）を焼き込む。
    ;; 前景色を明示しない項目（git-info / file-size / buffer-line-count など）は
    ;; この色を引き継ぐため、dark では背景に黒文字が載って読めなくなる。
    ;; 現在の default の前景色へ追従させる。
    (when (facep 'nano-modeline--empty-face)
      (set-face-attribute 'nano-modeline--empty-face nil
                          :inherit nil
                          :foreground (face-attribute 'default :foreground nil t)))
    (when (facep 'ui-nano-modeline-secondary)
      (if (eq frame-background-mode 'dark)
          (set-face-attribute 'ui-nano-modeline-secondary nil
                              :inherit nil
                              :foreground (ui-nano-modeline--dark-foreground))
        (set-face-attribute 'ui-nano-modeline-secondary nil
                            :inherit 'nano-faded
                            :foreground 'unspecified))))

  ;; 右側（時刻・エンコーディング・行列位置）は nano-modeline の secondary フェイスで
  ;; 描かれ、既定では `nano-faded' を継承する。dark では nano-faded (#677691) が
  ;; ヘッダー背景 nano-subtle (#434C5E) に近く、明暗比が約 1.9:1 しかなく読みにくい。
  ;; nano-faded は本文の注釈にも使うため変更せず、モードライン専用のフェイスを挟む。
  (defface ui-nano-modeline-secondary
    '((t :inherit nano-faded))
    "nano-modeline の右側表示用フェイス。dark では前景色を明るくする。"
    :group 'ui-nano-modeline)

  ;; git-info / file-size / buffer-line-count は primary フェイスで描かれる。
  ;; light / dark とも右側（secondary）と同じ色にそろえる。
  (defface ui-nano-modeline-primary
    '((t :inherit ui-nano-modeline-secondary))
    "nano-modeline の左側の補助情報用フェイス。右側と同じ色にする。"
    :group 'ui-nano-modeline)

  (with-eval-after-load 'nano-modeline
    ;; 既定値の status-**-active は nano-popout-i が未定義だと nil を含む
    ;; （`face' 型に合わない）。setopt の型検証で警告が出るため nil を除く。
    (setopt nano-modeline-faces
            (append '((secondary-active . (ui-nano-modeline-secondary))
                      (primary-active . (ui-nano-modeline-primary)))
                    (mapcar (lambda (entry)
                              (cons (car entry) (delq nil (copy-sequence (cdr entry)))))
                            (seq-remove (lambda (entry)
                                          (memq (car entry)
                                                '(secondary-active primary-active)))
                                        nano-modeline-faces))))
    (ui-nano-modeline-sync-faces))

  (provide 'ui-nano-modeline-items)

  ;; 設計目標:
  ;;
  ;; - 決定論的な初期化
  ;; - デーモンに対して安全なフレーム処理
  ;; - テーマ再読込に対する安全性
  ;; - Emacs 30 / 31 への前方互換性

  ;;   - 'none: 何もしない（Emacs の既定のテーマのまま）。

  ;;   'doom や 'none ではパレットのロードは強制されない。`my:use-ui' が nano 系のときだけ
  ;;   パレットをロードして nano を適用する。

  ;;   パレットのファイルが無い場合に備えるためである。
  ;;
  ;;   ui-nano-palette の正式なロード経路:
  ;;   ui-nano-palette の初回ロードを起動する責任を持つのは ui-theme.el のみである。
  ;;   ui-theme は常に my:modules に含まれるが、下記の
  ;;   `(require 'ui-nano-palette nil t)' は nano 系のときだけ実行される。
  ;;   ui-nano-modeline.el と dev-term.el は `with-eval-after-load' で反応するのみで、
  ;;   ui-nano-palette を直接 require してはならない。

  ;;; Code:

  ;; バイトコンパイル時の警告を避けるための宣言（実体は各パッケージ／ライブラリ側にある）。
  (declare-function nano-dark "nano-theme-support")
  (declare-function nano-light "nano-theme-support")

    (require 'leaf))

  (defun ui-theme-nano-p ()
    "Return non-nil when `my:use-ui' selects a bundle that uses the nano theme.
  That is `nano' or `doom-nano'."
    (and (memq (bound-and-true-p my:use-ui) '(nano doom-nano)) t))

  (defun ui-theme-doom-p ()
    "Return non-nil when `my:use-ui' is `doom', which uses a doom-themes theme.
  With `none' the default theme is kept."
    (eq (bound-and-true-p my:use-ui) 'doom))

  ;; NOERROR 形式（第 3 引数 t）なのは、ファイルが見つからなくても起動を止めないため。
  ;; nano 系の UI のときだけロードする（冒頭の Commentary を参照）。
  (when (ui-theme-nano-p)
    (require 'ui-nano-palette nil t)

    ;; パレットのロード直後に、実際のパレット色をテーマへ反映させるためフェイスを再適用する。
    ;; すでにロード済みならここで即座に実行され、未ロードならロードされた時点で実行される。
    (with-eval-after-load 'ui-nano-palette
      (when (fboundp 'ui-theme--apply-faces)
        (ui-theme--apply-faces))))

    "UI theme configuration."
    :group 'ui)

  (defcustom ui-theme-enable-p t
    "Enable UI theme configuration."
    :type 'boolean
    :group 'ui-theme)

  (defcustom ui-theme-doom-theme 'doom-one
    "The doom-themes theme to load when `my:use-ui' is `doom'.
  Any theme that doom-themes provides, for example `doom-one',
  `doom-one-light' or `doom-nord'."
    :type 'symbol
    :group 'ui-theme)

  (defun ui-theme--color (symbol &optional fallback)
    "パレット変数 SYMBOL の色を返す。未定義・空なら FALLBACK（省略時は nil）を返す。
  色の正本は `ui-nano-palette'。ここに色の固定値を重ねて持たないための窓口。
  `ui-nano-palette-color-*' には FALLBACK を渡さない（既定値は `ui-nano-palette' だけが持つ）。

    "Apply ARGS to FACE when FACE exists."
    (when (facep face)

    (when (fboundp 'ui-nano-palette-apply-faces)
      (ui-nano-palette-apply-faces))

    (ui-theme--safe-face 'bold :weight 'bold)
    (ui-theme--safe-face 'italic :slant 'italic)

    ;; 以下の固定色は明るい背景用。dark では nano-theme 側に任せる。

      (let ((subtle (ui-theme--color 'ui-nano-palette-color-subtle))

          (ui-theme--safe-face 'electric-pair-overlay-face :background subtle))

           :background subtle

* ui/ui-bundles.el
:PROPERTIES:
:CUSTOM_ID: ui-bundles
:header-args:emacs-lisp: :tangle ../lisp/ui/ui-bundles.el
:END:

UI バンドル（doom、nano、doom-nano）を =core-switches= の登録簿へ登録するモジュールです。

=core-switches= は core 層にいるため、ui 層のモジュール名を知ってはなりません。そこで名前を知っている
ui 層の側が、 =core-switches-register= で「この選択肢は、このファイルのこの関数で有効化する」と登録します。
登録は =autoload= を伴うので、バンドルのファイルは =my:use-ui= で選ばれたときにだけロードされます。
バンドル本体（ =ui-doom-modeline= 、 =ui-nano-modeline= 、 =ui-doom-nano-modeline= ）は、
どのリストにも載らず、ここで登録した =autoload= 経由でだけロードされます。

#+NAME: ui-bundles-el
#+begin_src emacs-lisp
  ;;; ui/ui-bundles.el --- UI バンドルの登録 -*- lexical-binding: t; -*-
  ;;
  ;; Copyright (c) 2021-2026
  ;; Author: YAMASHITA, Takao
  ;; License: GNU GPL v3 or later
  ;;
  ;; Category: ui
  ;;
  ;;; Commentary:
  ;;
  ;; UI バンドルを `core-switches' の登録簿へ登録する。
  ;;
  ;; 本ファイルの責務:
  ;; - doom、nano、doom-nano の 3 つを `core-switches-register' で登録する
  ;;   （エントリポイントは `my/ui-enable-doom'、`my/ui-enable-nano'、`my/ui-enable-doom-nano'）
  ;; - `my/ui-nano-modeline-toggle' の autoload を宣言する
  ;;
  ;; core-switches は core 層にあり、ui 層のモジュール名を知らない。名前を知る側が登録する。
  ;; バンドル本体は autoload からのみロードされ、`my:modules' にも `my:modules-extra' にも載らない。
  ;;; Code:

  (require 'core-switches)

  (core-switches-register 'ui 'doom 'my/ui-enable-doom
                          "ui/ui-doom-modeline" "Doom UI" "doom-modeline")
  (core-switches-register 'ui 'nano 'my/ui-enable-nano
                          "ui/ui-nano-modeline" "Nano UI" "nano-modeline")

  ;;
  ;; Copyright (c) 2021-2026
  ;; Author: YAMASHITA, Takao
  ;; License: GNU GPL v3 or later
  ;;
  ;; Category: ui
  ;;
  ;;; Commentary:
  ;;
  ;; nano の light / dark の切替、選んだ variant の保存、起動時の復元。
  ;;
  ;; 本ファイルの責務:
  ;; - `my/ui-theme-nano-switch' で nano-light / nano-dark を切り替え、選択を保存する
  ;;   （`my:use-ui' が nano 系でなければ user-error）
  ;; - 前回の選択を `ui-theme-variant-restore' で復元する（ui-theme.el の初期化の最後に呼ばれる）
  ;; - `ui-theme-variant-follow-system' が非 nil なら、macOS の外観（ライト／ダーク）に追従する。
  ;;   起動時と `ns-system-appearance-change-functions' で `my/ui-theme-nano-switch' を呼ぶ。
  ;;   この場合、保存した選択より OS の外観を優先する。手動の切替は次に外観が変わるまでの上書きになる。
  ;;   デーモンでは `ns-system-appearance' が最初のフレームができるまで決まらないため、
  ;;   `after-make-frame-functions' から 1 tick 遅らせて合わせる（AGENTS.md の Daemon-mode pattern）
  ;;; Code:

  (defgroup ui-theme-variant nil
    "nano の light / dark の切替と保存。"
    :group 'faces)

  (defcustom ui-theme-variant-follow-system t
    "非 nil なら、macOS の外観（ライト／ダーク）に nano の variant を合わせる。"
    :type 'boolean
    :group 'ui-theme-variant)

  (defvar ns-system-appearance)
  (defvar ns-system-appearance-change-functions)

  (declare-function nano-dark "nano-theme-support")
  (declare-function nano-light "nano-theme-support")

  (defvar ui-theme-variant--file
    (expand-file-name "nano-variant" my:d:var)
    "最後に選んだ nano の variant（light / dark）を保存するファイル。")

  (defun ui-theme-variant--save (variant)
    "VARIANT を `ui-theme-variant--file' に保存する。失敗しても無視する。"
    (ignore-errors
      (with-temp-file ui-theme-variant--file
        (insert (symbol-name variant) "\n"))))

  (defun ui-theme-variant-restore ()
    "保存された variant が dark なら nano-dark を適用する。"
    (when (file-readable-p ui-theme-variant--file)
      (when (equal (string-trim
                    (with-temp-buffer
                      (insert-file-contents ui-theme-variant--file)
                      (buffer-string)))
                   "dark")
        (my/ui-theme-nano-switch 'dark)))
    (ui-theme-variant--follow-setup))

  (defun ui-theme-variant--apply-system (&optional appearance)
    "APPEARANCE（`light' / `dark'。省略時は `ns-system-appearance'）に variant を合わせる。
  `ui-theme-variant-follow-system' が nil、nano が無効、APPEARANCE が不明のときは何もしない。"
    (let ((app (or appearance (bound-and-true-p ns-system-appearance))))
      (when (and ui-theme-variant-follow-system
                 (memq app '(light dark))
                 (fboundp 'ui-theme-nano-p)
                 (ui-theme-nano-p))
        (my/ui-theme-nano-switch app))))

  (defun ui-theme-variant--follow-setup ()
    "OS の外観への追従を設定する。外観の変化時と、現在の外観での初期化を登録する。
  デーモンでは最初のフレームができてから、1 tick 遅らせて合わせる。"
    (when ui-theme-variant-follow-system
      (add-hook 'ns-system-appearance-change-functions
                #'ui-theme-variant--apply-system)
      (if (daemonp)
          (add-hook 'after-make-frame-functions
                    (lambda (_frame)
                      (run-with-timer 0 nil #'ui-theme-variant--apply-system)))
        (ui-theme-variant--apply-system))))

  ;;;###autoload
  (defun my/ui-theme-nano-switch (variant)
    "nano テーマを VARIANT（`light' または `dark'）に切り替える。
    `nano-light' / `nano-dark' は `enable-theme' 経由で `nano' を再有効化し、
    `default-frame-alist' も更新する。`load-theme' で nano-light 等を重ねない。"
    (interactive
     (list (intern (completing-read "nano variant: " '("light" "dark") nil t))))
    (when (and (fboundp 'ui-theme-nano-p) (not (ui-theme-nano-p)))
      (user-error "nano theme is not enabled (see `my:use-ui')"))
    (require 'nano-theme-support)
    (if (eq variant 'dark) (nano-dark) (nano-light))
    (ui-theme-variant--save variant))

  (provide 'ui-theme-variant)
  ;;; ui/ui-theme-variant.el ends here

** ui-health-modeline.el — Session Status Display

モードラインに、バッファ数・プロセス数・Eglot が管理するバッファ数を出すだけのモジュール
です。ただし、書き方には注意が必要です。

*eglot が無い環境でも壊さない*

=featurep= と =fboundp= の二重チェックで、eglot が存在しない環境でのエラーを防いでいます。
eglot が読み込まれていて接続が無いときは、管理下のバッファが 0 個なので =LSP:0= と表示されます。

*軽さが正義*

この関数はモードラインが更新されるたびに呼ばれます。つまり、ほぼ全てのコマンドの後に走ると
思ってください。重い処理を書けば、その分だけ Emacs 全体がもたつきます。

=length= による O(n) の走査は、バッファ数・プロセス数の規模なら許容範囲です。それ以上のことを
ここでやりたくなったら、いったん立ち止まってください。
#+begin_src emacs-lisp
  ;;; ui/ui-health-modeline.el --- セッションの健全性インジケータ -*- lexical-binding: t; -*-
  ;;
  ;; Copyright (c) 2021-2026
  ;; Author: YAMASHITA, Takao
  ;; License: GNU GPL v3 or later
  ;;
  ;; Category: ui
  ;;
  ;;; Commentary:
  ;;
  ;; モードラインに軽量なセッション健全性インジケータを表示する。
  ;;
  ;; 表示する指標:
  ;;
  ;;  - バッファ数
  ;;  - 実行中プロセス数
  ;;  - Eglot が管理するバッファ数
  ;;
  ;; これらの指標により、重い計測機構を持ち込むことなく、Emacs セッションの負荷と
  ;; バックグラウンド活動を即座に把握できる。

  ;;; Code:

  (eval-when-compile (require 'leaf) (require 'subr-x))

  (defgroup ui-health-modeline nil
    "Mode-line indicators for session health."
    :group 'mode-line)

  (defcustom ui-health-show-buffers-p t
    "If non-nil, display the number of live buffers."
    :type 'boolean
    :group 'ui-health-modeline)

  (defcustom ui-health-show-processes-p t
    "If non-nil, display the number of running processes."
    :type 'boolean
    :group 'ui-health-modeline)

  (defcustom ui-health-show-eglot-p t
    "If non-nil, display the number of Eglot-managed buffers."
    :type 'boolean
    :group 'ui-health-modeline)

  (defvar ui-health--mode-line-string ""
    "Mode-line string containing session health indicators.")

  (defun ui-health--buffers-count ()
    "Return the number of live buffers."
    (length (buffer-list)))

  (defun ui-health--processes-count ()
    "Return the number of active processes."
    (length (process-list)))

  (defun ui-health--eglot-count ()
    "Return the number of buffers currently managed by Eglot.

  Returns nil when Eglot is not loaded or unavailable."
    (when (and ui-health-show-eglot-p
               (featurep 'eglot)
               (fboundp 'eglot-managed-buffers))
      (length (eglot-managed-buffers))))

  (defun ui-health--update ()
    "Update the mode-line session health indicators."
    (setq ui-health--mode-line-string
          (concat
           (when ui-health-show-buffers-p
             (format " B:%d" (ui-health--buffers-count)))
           (when ui-health-show-processes-p
             (format " P:%d" (ui-health--processes-count)))
           (when-let* ((n (ui-health--eglot-count)))
             (format " LSP:%d" n))))
    (force-mode-line-update))

  (add-hook 'buffer-list-update-hook #'ui-health--update)
  (add-hook 'process-status-change-hook #'ui-health--update)

  (with-eval-after-load 'eglot
    (add-hook 'eglot-managed-mode-hook #'ui-health--update)

* ui/ui-keys.el
:PROPERTIES:
:CUSTOM_ID: ui-keys
:header-args:emacs-lisp: :tangle ../lisp/ui/ui-keys.el
:END:

キー操作のモジュールです。リーダーキーの基盤、リーダーへの割り当て、dirvish のキー、リーダー外のグローバルキーを
1 つのファイルにまとめています。

公開 API の名前（ =ui-keymap-bind= 、 =ui-leader-key= 、 =ui-leader-map= 、 =ui-leader-*-map= ）は変えていません。
=personal/= 、 =dev/= 、 =utils/= 、 =orgx/= が参照しているためです。そのため、関数と変数の接頭辞はモジュール名
 =ui-keys= と一致しません。

=after-init-hook= には、 =ui-which-key--startup-init= 、 =ui-keymap--startup-init= 、 =ui-dirvish-keys--startup-init= 、
=ui-global-keys--startup-init= の順に登録します。 =personal/= と =dev/= 、 =utils/= の hook は depth 10 以降なので、このあとに走ります。
Hydra、which-key、キーボード変換も、キー操作の一部としてこのファイルに含めます。
ロード時の実行順は、リーダー基盤、キーボード変換の順です。 =after-init-hook= には which-key を先に登録します。

** リーダーキー基盤

なぜ =M-SPC= なのか

[[file:../svg/11_leader_keymap.svg]]

リーダーキーは =C-c SPC= から =M-SPC= に変更しました。3 打鍵が 2 打鍵になります。

=M-SPC= を選んだ理由は 3 つあります。素の Emacs で使われていないこと、macOS に横取り
されないこと、そして親指とスペースバーの和音として押しやすいことです。リーダーはすべての操作の
入口になるので、ここでの 1 打鍵の差は積み重なると大きく効きます。

*実装の要点*

=define-prefix-command= で、プレフィックス用のキーマップを変数として作っています。すでに
キーマップがあれば作り直さずに再利用するので、何度呼んでも安全です。

キーの割り当ては =keymap-set= を使います。 =define-key= + =kbd= の組み合わせを置き換える、
Emacs 29 以降の API です。

*サブマップの連鎖*

=M-SPC b l= は次のように解決されます。

: M-SPC b l → ui-leader-map["b"] → ui-leader-b-map["l"] → ibuffer

2 段の間接参照になっているので、 =b= 配下に何を足しても、他のグループには影響しません。

  	      ("-" text-scale-decrease)

** キーボード変換

=C-h= を =DEL= に読み替える変換です。オプションを変えると、変数ウォッチャーが再適用します。

#+NAME: ui-keys-keyboard
#+begin_src emacs-lisp

  (defgroup ui-keyboard nil
    "Keyboard translation helpers."
    :group 'ui)

  (defcustom ui-keyboard-map-c-h-to-del t
    "If non-nil, translate C-h to DEL (Backspace)."
    :type 'boolean
    :group 'ui-keyboard)

  (defvar ui-keyboard--translation-active-p nil
    "Non-nil when the C-h → DEL translation is currently active.")

  (defun ui-keyboard--apply ()
    "Apply or remove keyboard translations according to configuration."
    (cond
     ((and ui-keyboard-map-c-h-to-del
           (not ui-keyboard--translation-active-p))
      ;; define-key + kbd → keymap-set（Emacs 29 以降の API）
      ;; key-translation-map は通常のキーマップであり、keymap-set に対応している。
      (keymap-set key-translation-map "C-h" "DEL")
      (setq ui-keyboard--translation-active-p t))
     ((and (not ui-keyboard-map-c-h-to-del)
           ui-keyboard--translation-active-p)
      (keymap-set key-translation-map "C-h" nil)
      (setq ui-keyboard--translation-active-p nil))))

  (ui-keyboard--apply)

  ;; watcher は無名 lambda ではなく名前付き defun にする。
  ;; add-variable-watcher に渡した無名 lambda は remove-variable-watcher で除去できず、
  ;; describe-function で検査もできず、eval-buffer をまたいだ重複登録に対しても安全でない。
  ;; 副作用を伴う登録は名前付き関数で行う、というプロジェクト全体の規約でもある。
  ;;
  (defun ui-keyboard--watcher (_sym _newval _op _where)
    "Variable watcher: re-apply keyboard translation when the option changes."
    (ui-keyboard--apply))

  (add-variable-watcher 'ui-keyboard-map-c-h-to-del #'ui-keyboard--watcher)

** リーダーへの割り当て

キーバインドの設計では、「よく使うものほど短く」を徹底しています。方針は次の 3 つです。

| 方針 | 例 |
|----------+-----|
| リーダーを短縮する | =C-c SPC= → =M-SPC= （−1 打鍵） |
| 高頻度コマンドを最上位に昇格する | =M-SPC RET= = consult-buffer（5→3 打鍵） |
| より短い全体キーが存在する場合はバインドを削除する | =next-buffer= は =s-→= （2 打鍵）の方が短い |

3 つ目が見落とされがちです。バインドは増やすほど良いわけではありません。既に短い経路がある
コマンドをリーダー配下にも置くと、覚えることが増えるだけで、実際には使われません。

なぜ =ui-keymap-bind= がシンボルを受け取るのか

=ui-keymap-bind= は =map= をキーマップそのものではなく、シンボルとして受け取ります。そして
=keymap-set= を呼ぶ前に、 =keymapp= で本当にキーマップかどうかを確認します。

この一手間のおかげで、リーダーキー基盤がまだ定義されていない状況でも、エラーで停止せずに
安全に動きます。削除安全性の考え方を、キーバインドの層でも守っている形です。

** dirvish バッファのキー

dirvish のバッファ内のキー割り当てです。キーの割り当ては ui 層の仕事なので、dirvish の本体の設定（ =core-dired.el= ）ではなく
ui 層に置いています。 =o= 、 =O= 、 =z= 、 =q= などの割り当てと、view-mode バッファを右側のサイドウィンドウに出す設定を持ちます。

登録は =after-init-hook= で行い、その中で =with-eval-after-load 'dirvish= を使います。

#+NAME: ui-keys-dirvish
#+begin_src emacs-lisp

  ;; バイトコンパイル時の警告を避けるための宣言（実体は各パッケージ／ライブラリ側にある）。
  (declare-function dired-find-file-other-window "dired")
  (declare-function dired-get-filename "dired")

  (defun ui-dirvish-keys--open-read-only ()
    "Open file-at-point in another window as read-only.
  Combines `dired-find-file-other-window' with `read-only-mode'
  so the dirvish session remains intact for continued browsing."
    (interactive)
    (dired-find-file-other-window)
    (read-only-mode 1))

  ;; `z' — dirvish 一覧を保持したまま file-at-point を別ウィンドウに RO 表示する。
  ;;
  ;; 上の `ui-dirvish-keys--open-read-only'（"O" にバインド）は
  ;; `dired-find-file-other-window' + `read-only-mode' の組み合わせだが、これは
  ;; 「編集可能なバッファを開いてから明示的に read-only にする」経路であり、
  ;; view-mode 特有の挙動（"q" で閉じてウィンドウ配置ごと元に戻る、末尾に
  ;; カーソルを置いても編集は一切できない、等）を持たない。
  ;;
  ;; `view-file-other-window' は最初から view-mode で開くため、「素早く中身を
  ;; 確認して q で戻る」用途に向いている。dired 標準の "v"（`dired-view-file'）は
  ;; カレントウィンドウを置き換えてしまい、dirvish の一覧が隠れてしまうため、
  ;; その代替として "z" に別ウィンドウ版を割り当てる。
  (defun ui-dirvish-keys--view-file-other-window ()
    "View file-at-point read-only in another window via `view-file-other-window'.
  Unlike `dired-view-file' (bound to the dired-native \"v\"), this keeps the
  current dirvish listing window intact instead of replacing it, and opens
  the target buffer directly in `view-mode' (not merely `read-only-mode'
  layered on top of a normal buffer)."
    (interactive)
    (view-file-other-window (dired-get-filename nil t)))

  ;; view-mode バッファの出現位置を固定する。
  ;;
  ;; `view-file-other-window' は内部的に `switch-to-buffer-other-window' →
  ;; `display-buffer' を通るため、`display-buffer-alist' の対象になる。指定が
  ;; ないと分割方向がフレームの状態（既存ウィンドウ数・フォーカス位置）に依存して
  ;; 毎回揺れる。フレーム最大化運用（ウィンドウ分割の目視手がかりが少ない）と
  ;; 特に相性が悪いため、常に右側の専用サイドウィンドウに固定する。
  ;;
  ;; `dedicated . t' により、view-mode を抜けた後（"q" で `view-mode' が
  ;; バッファを kill/quit する際）にこのサイドウィンドウ自体も片付く。
  (add-to-list 'display-buffer-alist
               '((lambda (buf _action)
                   (with-current-buffer buf (derived-mode-p 'view-mode)))
                 (display-buffer-reuse-window display-buffer-in-side-window)
                 (side . right)
                 (window-width . 0.5)
                 (dedicated . t)))

  (defun ui-dirvish-keys--startup-init ()
    "Register the dirvish key bindings once dirvish is loaded."
    (with-eval-after-load 'dirvish
      (when (boundp 'dirvish-mode-map)
        (keymap-set dirvish-mode-map "TAB" #'dirvish-subtree-toggle)
        (keymap-set dirvish-mode-map "i"   #'dirvish-subtree-toggle)
        (keymap-set dirvish-mode-map "o"   #'dired-find-file-other-window)
        (keymap-set dirvish-mode-map "O"   #'ui-dirvish-keys--open-read-only)
        ;; "z" は dired 標準では `dired-do-compress'（マークしたファイルの圧縮）に
        ;; 割り当てられている。dirvish セッションでは頻度が低いと判断し、
        ;; view-mode での別ウィンドウ RO 表示に明示的に上書きする。
        (keymap-set dirvish-mode-map "z"   #'ui-dirvish-keys--view-file-other-window)
        (keymap-set dirvish-mode-map "q"   #'dirvish-quit))))

  (add-hook 'after-init-hook #'ui-dirvish-keys--startup-init)

  (defun my/keyboard-quit-dwim ()
    "Do-What-I-Mean behaviour for `keyboard-quit'.

    The generic `keyboard-quit' does not close the minibuffer when point is
    in another window.  This command handles, in order:

    - an active region: deactivate it (via `keyboard-quit');
    - the *Completions* buffer being selected: close its window;
    - a minibuffer open but not focused: abort it;
    - otherwise: behave as `keyboard-quit'."
    (interactive)

     ((region-active-p)
      (keyboard-quit))
     ((derived-mode-p 'completion-list-mode)
      (delete-completion-window))
     ((> (minibuffer-depth) 0)
      (abort-recursive-edit))
     (t
      (keyboard-quit))))

  ;;
  ;; Copyright (c) 2021-2026

  ;; Category: ui

  ;;; Commentary:
  ;;
  ;; 控えめな視覚補助: パルス、TODO ハイライト、括弧の色分け、インデントガイド。
  ;;
  ;; 本ファイルの責務:
  ;; - ナビゲーション系コマンドで一時的にカーソル位置を示す pulsar を設定する
  ;;   （起動時間短縮のため、有効化自体は emacs-startup-hook 経由でアイドル
  ;;   タイマーに遅延させる）
  ;; - 色分けフェイス付きの hl-todo を有効化する（TODO、FIXME、NOTE、HACK）
  ;; - 括弧の深さを色分けする rainbow-delimiters を `prog-mode' で有効化する
  ;; - インデントの視覚ガイドとして indent-bars を `prog-mode' で有効化する
  ;;
  ;; ロード経路:
  ;;
  ;;   本モジュールは `my:modules' に含まれない。表示に関する任意の選択として扱われ、
  ;;   personal/user.el が `my:modules-extra' に追加する。personal/user.el が無い場合
  ;;   （personal オーバーレイの無い新規チェックアウトなど）、本モジュールはロードされず、
  ;;   その他の設定は完全に機能したままとなる。
  ;;
  ;;; Code:

  ;; バイトコンパイル時の警告を避けるための宣言（実体は各パッケージ／ライブラリ側にある）。
  (defvar pulsar-pulse-functions)

  ;; pulsar は起動プロファイル計測で 16〜22ms を占めていた。leaf の :custom /
  ;; :config は暗黙に :require t を伴うため、モジュールのロード時に pulsar が
  ;; 即時 require されていたことが原因である。pulsar はナビゲーション系
  ;; コマンドの後にカーソル位置を一瞬光らせるだけの視覚効果であり、起動完了の

  ;; パターンと同じ）。パッケージの導入自体は `:straight t' がここで保証する

    (require 'pulsar)
    (setopt pulsar-delay 0.04
            pulsar-pulse t
            pulsar-face  'pulsar-generic)

      (add-to-list 'pulsar-pulse-functions cmd)))

    "起動完了後、アイドルタイマーで `ui-visual-aids--activate-pulsar' を呼ぶ。"

    :hook (prog-mode-hook . rainbow-delimiters-mode))

#+begin_src emacs-lisp
  ;;; ui/ui-macos.el --- macOS 固有の UI 調整 -*- lexical-binding: t; -*-
  ;;
  ;; Copyright (c) 2021-2026
  ;; Author: YAMASHITA, Takao
  ;; License: GNU GPL v3 or later
  ;;
  ;; Category: ui
  ;;
  ;;; Commentary:
  ;;
  ;; グラフィカルフレーム向けの macOS 固有 UI 調整。
  ;;
  ;; 本ファイルの責務:
  ;; - macOS で透過タイトルバー（`ns-transparent-titlebar'）を有効化する
  ;; - ネイティブな macOS タイトルバー描画のため `use-title-bar' を設定する
  ;; - Emacs 31 の `system-taskbar-mode'（Dock）を有効化する
  ;; - コンパイル中のアイドルスリープを抑止し、スリープ前に TRAMP の接続を切る
  ;; - すべての設定を `(eq system-type 'darwin)' でガードする
  ;;
  ;; ロード経路:
  ;;
  ;;   本モジュールは `my:modules' に含まれない。macOS 専用の任意の表示モジュールであり、
  ;;   personal/user.el が `my:modules-extra' に追加する。
  ;;   macOS 以外のホストでも、これが無いまま設定は機能する。
  ;;; Code:

  (eval-when-compile (require 'leaf))

  (when (eq system-type 'darwin)
    (add-to-list 'default-frame-alist '(ns-transparent-titlebar . t))
    (add-to-list 'default-frame-alist '(use-title-bar . t)))

  ;; ---------------------------------------------------------------------------
  ;; Emacs 31: システム統合（macOS 10.9 以降）
  ;; ---------------------------------------------------------------------------
  ;; `system-taskbar-mode' は Dock アイコンを制御する。バッジ数、プログレスバーの重ね表示、
  ;; 「要注意」のバウンドなど。`system-sleep' は長時間ジョブの実行中にアイドルスリープを
  ;; 抑止し、スリープ前／復帰後のフックを提供する。
  ;;
  ;; いずれも、本設定が想定するコンパイル負荷の高い作業（ネイティブコンパイル、Emacs の
  ;; ソースビルド）で有用である。ビルド途中でマシンがスリープしなくなり、リモート（TRAMP）
  ;; 接続はスリープ前に切断されるため、復帰時にハングしない。
  ;;
  ;; system-taskbar-mode のバックエンド判定（system-taskbar--set-back-end）は
  ;; (boundp 'ns-version-string) というビルド種別のチェックであり、実行時の
  ;; フレーム有無とは無関係。'ns バックエンドの system-taskbar--enable は
  ;; フレームの存在を一切確認せず無条件に t を返す実装になっている
  ;; （system-taskbar.el 本体で確認済み）。そのため display-graphic-p による
  ;; daemon 向けの遅延ガードは不要と判断し、単純な無条件呼び出しに戻した。
  (when (and (eq system-type 'darwin) (fboundp 'system-taskbar-mode))
    (system-taskbar-mode 1))

  (defvar ui-macos--sleep-block-token nil
    "Token returned by `system-sleep-block-sleep', or nil when not blocking.")

  (defun ui-macos--block-sleep-during-compile (_proc)
    "Inhibit system idle sleep while a compilation is running.
  Called from `compilation-start-hook'.  The display is allowed to sleep."
    (when (and (fboundp 'system-sleep-block-sleep)
               (null ui-macos--sleep-block-token))
      (setq ui-macos--sleep-block-token
            (system-sleep-block-sleep "Emacs: compiling" t))))

  (defun ui-macos--unblock-sleep-after-compile (_buffer _status)
    "Release the sleep block once compilation finishes.
  Called from `compilation-finish-functions'."
    (when (and ui-macos--sleep-block-token
               (fboundp 'system-sleep-unblock-sleep))
      (system-sleep-unblock-sleep ui-macos--sleep-block-token)
      (setq ui-macos--sleep-block-token nil)))

  ;; Emacs 31 の `tramp-cleanup-bufferless-connections' は、対応するバッファが無い
  ;; 接続だけを破棄する。スリープ前に必要なのは、リモートバッファを開いたままの接続も
  ;; 含めて切ることなので、ここでは従来どおり `tramp-cleanup-all-connections' を使う。
  (defun ui-macos--handle-sleep-event (event)
    "Close remote connections on EVENT when the system is about to sleep.
  Pre-sleep handlers get only a few seconds and must not signal, hence the
  `condition-case'.  Connections are re-established lazily on next access."
    (condition-case nil
        (when (and (fboundp 'sleep-event-state)
                   (eq (sleep-event-state event) 'pre-sleep)
                   (fboundp 'tramp-cleanup-all-connections))
          (tramp-cleanup-all-connections))
      (error nil)))

  (when (and (eq system-type 'darwin) (require 'system-sleep nil t))
    (add-hook 'compilation-start-hook     #'ui-macos--block-sleep-during-compile)
    (add-hook 'compilation-finish-functions #'ui-macos--unblock-sleep-after-compile)
    (add-hook 'system-sleep-event-functions #'ui-macos--handle-sleep-event))

  (provide 'ui-macos)
  ;;; ui/ui-macos.el ends here
#+end_src
