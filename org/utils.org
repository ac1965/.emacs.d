  ;;; utils/utils-buffers.el --- バッファ管理の補助 -*- lexical-binding: t; -*-
  ;;
  ;;
  ;; Copyright (c) 2021-2026
  ;; Author: YAMASHITA, Takao
  ;; License: GNU GPL v3 or later
  ;;
  ;; Category: utils
  ;;
  ;;; Commentary:
  ;;
  ;; バッファ管理: 保存確認を伴うスマートな kill。
  ;;
  ;; 本ファイルの責務:
  ;; - `my/smart-kill-buffer': 変更済みファイルバッファを kill する前に保存を確認し、
  ;;   ユーザーの回答にかかわらず kill する
  ;; - 上記を `ui-leader-map' の k （M-SPC k）へ自分で割り当てる
  ;;   （`utils-buffers--bind-keys'）
  ;;
  ;; キー割り当てに関する注記:
  ;; ui-keys.el が `my/smart-kill-buffer' を直接参照すると、ui 層が utils 層に依存する
  ;; ことになり層の方向に反する。そのため割り当ては本モジュールが行う。
  ;; `ui-leader-map' は ui-keymap の `after-init-hook' で作られるので、hook は depth 90
  ;; で登録する。マップが無ければ何もしない。
  ;;; Code:

  (defun my/smart-kill-buffer (&optional buffer-or-name)
    "Kill BUFFER-OR-NAME after saving if needed."
    (interactive)
    (let ((buf (or (and buffer-or-name (get-buffer buffer-or-name)) (current-buffer))))
      (when (buffer-live-p buf)
        (when (and (buffer-modified-p buf) (buffer-file-name buf))
          (when (y-or-n-p (format "Save buffer %s? " (buffer-name buf)))
            (with-current-buffer buf (save-buffer))))
        (kill-buffer buf))))

  (defvar ui-leader-map)

  (defun utils-buffers--bind-keys ()
    "Bind `my/smart-kill-buffer' in `ui-leader-map' when the map exists."
    (when (and (boundp 'ui-leader-map) (keymapp ui-leader-map))
      (keymap-set ui-leader-map "k" #'my/smart-kill-buffer)))

  ;; depth 90: ui-keymap--startup-init （depth 0）がマップを作った後に走らせる。
  (add-hook 'after-init-hook #'utils-buffers--bind-keys 90)

  (provide 'utils-buffers)
  ;;; utils/utils-buffers.el ends here
#+end_src

  ;; 破棄してしまう。`dirvish-find-file' は既存セッション内で移動する — セッション内の
  ;; ディレクトリ進入にはこちらが正しい API である。
  ;;; Code:

  (defvar dired-mode-map)
  (declare-function dired-find-file "dired")
  (declare-function dired-get-file-for-visit "dired")

  (defun my/dired-get-file-or-dir ()
    "Visit file at point, or enter the directory in the current dirvish session."
    (interactive)
    (let ((file (dired-get-file-for-visit)))
      (if (file-directory-p file)
          (if (fboundp 'dirvish-find-file)
              (dirvish-find-file)
            (dired-find-file))
        (find-file file))))

  (with-eval-after-load 'dired
    (keymap-set dired-mode-map "RET" #'my/dired-get-file-or-dir))

  (provide 'utils-dired)
  ;;; utils/utils-dired.el ends here

** utils-ime.el — 入力ソース（IME）制御

日本語 IME をかな入力にしたままだと、リーダーキー =M-SPC= の後続打鍵
（ =b l= など）や =dired= / =magit-status= /ターミナル系バッファでのナビゲーション・
操作キー（ =RET= / =g= / =d= / =s= など）が IME に composed されて誤動作します。
本モジュールは =macism= を同期呼び出しして、次の契機で OS の入力ソースを
ASCII（英字）へ切り替えます。

- *リーダーキー押下時* — =ui-leader-key= を =menu-item= の =:filter= で包み、
  キー押下の瞬間に ASCII 化してから =ui-leader-map= を返します。プレフィックスキーの
  性質は保たれるので、 which-key と =describe-key= は従来どおり機能します。単発の
  ディスパッチ用途なので、復帰処理は行いません。
- *対象メジャーモード進入／離脱時* — =utils-ime-target-modes= （既定:
  =dired-mode= =magit-status-mode= =vterm-mode= =eshell-mode= =term-mode=
  =comint-mode= =prog-mode= 。 =prog-mode= は次の項目で説明します）に該当するバッファがカレントになった瞬間に ASCII へ切り替えます。
  そのモードから離れて対象外のバッファへ戻った瞬間には、入る直前の入力ソースへ
  自動で復帰します。 =buffer-list-update-hook= でカレントバッファの遷移を監視し、
  対象／非対象の境界をまたいだときだけ切替・復帰を行います。 =wdired-mode= は
  =dired-mode= 上のマイナーモードなので、 =dired-mode= を対象にするだけで
  wdired 編集中も自動的にカバーされます（個別のフックは要りません）。dirvish も同様に
  =dired-mode= が土台なので、自動的に対象になります。
- *プログラミング言語バッファ進入／離脱時*（追加分） — =utils-ime-target-modes=
  に =prog-mode= を加えることで、Python／Elisp／JS など、ほとんどの
  プログラミング言語メジャーモードが =derived-mode-p= 経由で自動的に対象になります。
  =org-edit-special=（ =C-c '= ）で開く Org Babel のソース編集バッファも、
  その言語自身のメジャーモード（ =python-mode= など）を持つ独立バッファです。
  そのため、この一行を足すだけで =buffer-list-update-hook= が既存の仕組みのまま検知し、
  新しいフックなしに自動的にカバーされます。 =org-mode= 自体は =prog-mode= の
  派生ではないので、本文（日本語散文）側は影響を受けません。なお、コード中に
  日本語コメントを書くときは、その間だけ手動で IME に切り替えてください。
- *ミニバッファ使用時* — =M-x= や =find-file= などでミニバッファに入った瞬間に
  ASCII へ切り替え、抜けた瞬間に入る直前の入力ソースへ復帰します
  （ =minibuffer-setup-hook= / =minibuffer-exit-hook= ）。
- *範囲選択中* — リージョンが有効になった瞬間に、日本語入力なら ASCII へ切り替え、解除したら
  元の入力ソースへ戻します（ =activate-mark-hook= / =deactivate-mark-hook= 。 =utils-ime-enable-region-p= ）。

フックの登録は =emacs-startup-hook= まで遅らせています。起動中のバッファの出入りで =macism= を同期で呼ぶと
（1 回約 40ms）、起動が約 80ms 遅くなるためです。

=macism= が見つからない環境（非 macOS ／未インストール）では、すべて no-op になります。
ASCII のソース ID は =utils-ime-ascii-source= で変更できます。既定は
=com.apple.keylayout.ABC= です（ =macism= を引数なしで実行すると現在値を確認できます）。
復帰先は固定値ではありません。対象コンテキストに入る直前に =macism= で取得した実際の
入力ソースを保存しておき、離脱時にそこへ戻します。

#+begin_src emacs-lisp
  ;;; utils/utils-ime.el --- IME（入力ソース）制御ユーティリティ -*- lexical-binding: t; -*-
  ;;
  ;; Copyright (c) 2021-2026
  ;; Author: YAMASHITA, Takao
  ;; License: GNU GPL v3 or later
  ;;
  ;; Category: utils
  ;;
  ;;; Commentary:
  ;;
  ;; macism を用いて OS の入力ソースを ASCII（英字）へ切り替え、対象コンテキストを
  ;; 離れたら元の入力ソースへ復帰するユーティリティ。
  ;;
  ;; 目的:
  ;; - リーダーキー（`ui-leader-key'）を押した瞬間に ASCII へ切り替え、後続の
  ;;   1 打鍵（`b l' など）が日本語 IME に composed されず、確実に届くようにする。
  ;;   （単発ディスパッチのため復帰は行わない）
  ;; - `utils-ime-target-modes' に列挙したメジャーモード（dired/dirvish、
  ;;   magit-status、vterm/eshell/term/comint、`prog-mode' 派生の全言語モード）
  ;;   に入ったら ASCII へ切り替え、`RET'/`g'/`d'/`s' などの操作キーの誤爆や、
  ;;   コード入力時の意図しない日本語 composition を防ぐ。そのバッファから
  ;;   離れたら、入る直前の入力ソースへ自動復帰する。`prog-mode' を含めたことで、
  ;;   Org Babel のソース編集バッファ（`org-edit-special' が開く独立バッファ、
  ;;   実体は言語自身のメジャーモード）も同じ仕組みで自動的にカバーされる。
  ;; - ミニバッファ（`M-x'、`find-file' など）に入ったら ASCII へ切り替え、抜けたら
  ;;   入る直前の入力ソースへ自動復帰する。
  ;;
  ;; 本ファイルの責務:
  ;; - `utils-ime-to-ascii': macism を同期呼び出しして ASCII 入力ソースへ切り替える。
  ;; - `utils-ime--install-leader': リーダーキーを `menu-item' の `:filter' で包み、
  ;;   キー押下時に ASCII 化してから `ui-leader-map' を返す。これにより which-key と
  ;;   `describe-key' の挙動はプレフィックスキーとして従来どおり維持される。
  ;; - `utils-ime--startup-init': 下の 4 種類のフックを `emacs-startup-hook' で登録する
  ;;   （起動中は macism を呼ばないため。起動時間を約 80ms 縮める）
  ;; - `utils-ime--track-buffer-change': `buffer-list-update-hook' 上で、カレント
  ;;   バッファが `utils-ime-target-modes' の境界をまたいだときに ASCII 化／復帰する。
  ;; - `utils-ime--minibuffer-setup' / `utils-ime--minibuffer-exit':
  ;;   `minibuffer-setup-hook' / `minibuffer-exit-hook' でミニバッファの
  ;;   ASCII 化／復帰を行う。
  ;; - `utils-ime--region-activate' / `utils-ime--region-deactivate':
  ;;   `activate-mark-hook' / `deactivate-mark-hook' で、範囲選択中だけ ASCII にし、
  ;;   選択を解除したら元の入力ソースへ戻す。
  ;;
  ;; 設計上の注意:
  ;; - macism は同期呼び出し（`call-process' の DESTINATION を nil とし待機）とする。
  ;;   非同期（DESTINATION 0）だと、切替完了前にユーザーが次を打鍵し IME に取られる
  ;;   競合が起きうるため。macism は数十 ms で完了するため実用上の遅延は無視できる。
  ;; - macism が見つからない環境（非 macOS／未インストール）では全て no-op。
  ;;   `utils-ime-available-p' で存在を確認し、なければ何もしない。
  ;; - `ui-keys' 未ロードでも安全なよう `with-eval-after-load' と `boundp' で防御する。
  ;; - `buffer-list-update-hook' は高頻度で発火するが、`utils-ime--in-target-p' で
  ;;   ガードし、対象／非対象の境界をまたいだ瞬間だけ macism を呼ぶため実用上のコスト
  ;;   は無視できる。
  ;; - `wdired-mode' はマイナーモードで、有効化してもバッファのメジャーモードは
  ;;   `dired-mode' のまま変わらない。よって `dired-mode' を対象に含めておくだけで
  ;;   wdired 編集中も自動的に ASCII のままになり、個別のフックは不要である。
  ;;
  ;;; Code:

  (defgroup utils-ime nil
    "Input-source (IME) control via the macism CLI."
    :group 'utils)

  (defcustom utils-ime-macism-executable (executable-find "macism")
    "Absolute path to the `macism' executable, or nil if unavailable.
    When nil, every command in this module is a no-op."
    :type '(choice (const :tag "Disabled" nil) file)
    :group 'utils-ime)

  (defcustom utils-ime-ascii-source "com.apple.keylayout.ABC"
    "Input-source ID used for ASCII (English) input.
    Common values: \"com.apple.keylayout.ABC\" or \"com.apple.keylayout.US\".
    Run `macism' with no arguments in a shell to print the current source ID."
    :type 'string
    :group 'utils-ime)

  (defcustom utils-ime-enable-leader-p t
    "If non-nil, switch to ASCII when the leader key is pressed."
    :type 'boolean
    :group 'utils-ime)

  (defcustom utils-ime-target-modes
    '(dired-mode magit-status-mode vterm-mode eshell-mode term-mode comint-mode
                 prog-mode)
    "Major modes that force ASCII input on entry and restore the previous
    input source on exit.  Matched via `derived-mode-p', so modes derived from
    any of these (e.g. `shell-mode' or `sql-interactive-mode' via `comint-mode',
    or `python-mode'/`emacs-lisp-mode'/etc. via `prog-mode') are covered too.
    Dirvish and `wdired-mode' need no separate entry: both are built on top of
    `dired-mode' and are therefore already covered.  Including `prog-mode' also
    covers Org Babel source-edit buffers opened via `org-edit-special' (C-c '),
    since each such buffer's major mode is the source language's own mode;
    `org-mode' itself does not derive from `prog-mode', so ordinary prose
    buffers are unaffected.  Typing Japanese comments inside a covered buffer
    still requires switching the OS input source back manually for that span."
    :type '(repeat symbol)
    :group 'utils-ime)

  (defcustom utils-ime-enable-mode-tracking-p t
    "If non-nil, switch to ASCII on entering a buffer whose major mode is in
    `utils-ime-target-modes', and restore the previous input source when the
    selected buffer leaves that set."
    :type 'boolean
    :group 'utils-ime)

  (defcustom utils-ime-enable-minibuffer-p t
    "If non-nil, switch to ASCII while the minibuffer is active (`M-x',
    `find-file', etc.) and restore the previous input source on exit."
    :type 'boolean
    :group 'utils-ime)

  (defcustom utils-ime-enable-region-p t
    "If non-nil, switch to ASCII while the region is active.
  範囲選択中の `M-w' / `C-w' などが日本語 IME に取られて Emacs に届かないのを防ぐ。
  選択を解除したら、切り替える直前の入力ソースへ戻す。"
    :type 'boolean
    :group 'utils-ime)

  (defvar utils-ime--in-target-p nil
    "Non-nil while the selected buffer is considered inside a target context.")

  (defvar utils-ime--saved-source nil
    "Input source captured immediately before entering a target-mode context.
    Nil when not currently inside one.")

  (defvar utils-ime--minibuffer-saved-source nil
    "Input source captured immediately before the minibuffer was activated.")

  (defvar utils-ime--region-saved-source nil
    "Input source captured immediately before the region became active.
  Nil when the source was already ASCII or the region is inactive.")

  (defun utils-ime-available-p ()
    "Return non-nil when macism can be used on this system."
    (and (eq system-type 'darwin)
         (stringp utils-ime-macism-executable)
         (file-executable-p utils-ime-macism-executable)))

  (defun utils-ime--current-source ()
    "Return the current OS input-source ID via macism, or nil on failure.
    Also nil when macism is unavailable."
    (when (utils-ime-available-p)
      (ignore-errors
        (with-temp-buffer
          (when (zerop (call-process utils-ime-macism-executable nil t nil))
            (string-trim (buffer-string)))))))

  (defun utils-ime--switch-to (source)
    "Switch the OS input source to SOURCE via macism.
    No-op when SOURCE is nil or macism is unavailable.  Synchronous by design
    (see Commentary)."
    (when (and source (utils-ime-available-p))
      (ignore-errors
        (call-process utils-ime-macism-executable nil nil nil source))))

  (defun utils-ime-to-ascii ()
    "Switch the OS input source to `utils-ime-ascii-source' via macism.
    No-op when macism is unavailable.  Errors are trapped so a failure never
    blocks command dispatch.  Synchronous by design (see Commentary)."
    (interactive)
    (utils-ime--switch-to utils-ime-ascii-source))

  (defun utils-ime--install-leader ()
    "Wrap the leader key so a keypress first switches to ASCII input.
    The leader map stays a real prefix (which-key/`describe-key' keep working)
    because the `menu-item' `:filter' returns `ui-leader-map' after the switch.
    This is a one-shot dispatch aid, so the previous input source is not saved
    or restored afterwards."
    (when (and utils-ime-enable-leader-p
               (boundp 'ui-leader-key) (stringp ui-leader-key)
               (boundp 'ui-leader-map) (keymapp ui-leader-map))
      (define-key global-map (kbd ui-leader-key)
                  `(menu-item "leader+ascii" ,ui-leader-map
                              :filter ,(lambda (_)
                                         (utils-ime-to-ascii)
                                         ui-leader-map)))))

  (defun utils-ime--target-buffer-p ()
    "Return non-nil if the current buffer's major mode is in
    `utils-ime-target-modes'."
    (apply #'derived-mode-p utils-ime-target-modes))

  (defun utils-ime--track-buffer-change ()
    "Switch to/from ASCII when the selected buffer crosses a target-mode
    boundary.  Installed on `buffer-list-update-hook', which fires on every
    buffer-selection change regardless of how it happened (`find-file',
    `quit-window', `other-window', `dirvish-quit', jumping from dired into
    `magit-status', ...).  Guarded by `utils-ime--in-target-p' so macism is only
    invoked exactly on entry/exit transitions, not on every command."
    (when (and utils-ime-enable-mode-tracking-p
               (not (minibufferp)))
      (let ((target (utils-ime--target-buffer-p)))
        (cond
         ((and target (not utils-ime--in-target-p))
          (setq utils-ime--saved-source (utils-ime--current-source))
          (setq utils-ime--in-target-p t)
          (utils-ime-to-ascii))
         ((and (not target) utils-ime--in-target-p)
          (setq utils-ime--in-target-p nil)
          (utils-ime--switch-to utils-ime--saved-source)
          (setq utils-ime--saved-source nil))))))

  (defun utils-ime--minibuffer-setup ()
    "Switch to ASCII when the minibuffer is activated, saving the prior source."
    (when utils-ime-enable-minibuffer-p
      (setq utils-ime--minibuffer-saved-source (utils-ime--current-source))
      (utils-ime-to-ascii)))

  (defun utils-ime--minibuffer-exit ()
    "Restore the input source saved by `utils-ime--minibuffer-setup'."
    (when (and utils-ime-enable-minibuffer-p utils-ime--minibuffer-saved-source)

設定を変えたあとの確認のうち、起動中の Emacs でしかできないものを、 =M-x= のコマンドにしています。
バッチで動く =make smoketest= （ =org/tools.org= の Appendix: smoketest.el）の続きにあたります。

| コマンド | 確認すること |
|---+---|
| =M-x my/smoketest-lsp= | 実際の言語サーバ（Python は =pyright-langserver= 、C は =clangd= 、Go は =gopls= ）に Eglot で接続し、サーバ自身の候補と、Eglot の候補への dabbrev の合流を確かめる |
| =M-x my/smoketest-terminal= | 子の =emacs -nw= を疑似端末で動かし、corfu-terminal が候補を文字で描くかを確かめる |
| =M-x my/smoketest-gui= | posframe、Corfu、popupinfo の子フレームが、実際に見える状態になるかを確かめる |
| =M-x my/smoketest= | 上の 3 つをまとめて実行する |

結果は =*Smoketest*= バッファに、 =PASS= 、 =FAIL= 、 =SKIP= の行で並びます。
サーバが無い環境や、グラフィカルでないフレームでは、該当の確認が =SKIP= になります。

*LSP の確認の作り*

- 言語ごとに、一時ディレクトリへ小さなファイルを作ります。Go には =go.mod= も置きます。
  試験用のファイルは専用のディレクトリに入れます。システムの一時ディレクトリ全体がプロジェクトの根になると、
  サーバがその全体を調べに行くためです。
- モードは設定せず、ファイルを開いたときに自動で選ばれたものを使います。 =auto-mode-alist= や
  tree-sitter の remap（ =core-treesit= ）まで含めて確かめるためです。Python は =python-ts-mode= 、
  C は =c-ts-mode= 、Go は =go-ts-mode= が選ばれるはずです。
- C の候補は、 =printf(const char * _Nonnull, ...)= のように、名前のあとに型の情報が付きます。
  このため候補の一致は「期待する名前で始まる」ことで判定します。
- サーバが補完要求に答えない場合は、合流の不具合と区別するため、以降の確認を =SKIP= にします。

*実行中は何も入力しないでください* 。GUI の確認は、キー入力を送ってタイマーとポップアップを待つため、
手でキーを打つと結果が狂います。

*安全のための作り*

- 端末の確認は、子の Emacs に *キー入力を送りません* 。子は =--eval= で自分専用のバッファを開き、
  そこで補完を呼びます。以前の実装は、起動後に文字を打ち込んでいたため、子が復元したファイル
  （ =init.el= ）に文字が入り込みました。復元されるファイルへ打つ作りは、二度と使わないでください。
- LSP の確認は、一時ファイルを作り、終了時にサーバを止めて消します。
- 確認できるのは、必要なフレームが作られ、見える状態になったかまでです。色や配置のような
  *見た目* そのものは、目で見て確かめてください。

  Each element is (LABEL FILE MODES TEXT PREFIX EXPECT EXECUTABLE

                                (list 'error (format "実際は %s" major-mode)))))

                      (eglot-autoshutdown t))

                (utils-diagnostics--smoketest-record
                 (format "LSP %s: 合流した CAPF が先頭にあり、Eglot 単体は消える" label)
                 (lambda () (and (functionp (car completion-at-point-functions))
                                 (not (memq 'eglot-completion-at-point
                                            completion-at-point-functions)))))
                ;; 合流した CAPF の前に、Eglot 単体が補完要求に答えるかを確かめる。答えないなら
                ;; 合流の不具合ではなくサーバ側の問題なので、以降は SKIP にする。

                        (format "LSP %s: サーバが補完要求に応答する" label)
                        (lambda ()
                          (let ((res (eglot-completion-at-point)))

                   (lambda ()
                     (if (not server-ok)
                         '(skip . "サーバが補完要求に応答しない")
                       (utils-diagnostics--smoketest-wait
                        (lambda ()
                          (let ((res (funcall (car completion-at-point-functions))))
                            (and res (cl-some (lambda (c) (string-prefix-p expect c))
                                              (all-completions prefix (nth 2 res))))))
                        20))))

                (when-let* ((server (eglot-current-server)))

            (kill-buffer buf)

* utils/utils-scratch.el
:PROPERTIES:
:CUSTOM_ID: utils-scratch
:header-args:emacs-lisp: :tangle ../lisp/utils/utils-scratch.el
:END:

=*scratch*= バッファのライフサイクルを管理し、追加の scratch バッファを作るコマンドを提供します。

以前は =*scratch*= の再生成が 2 か所にありました。 =core/core-buffers.el= は =kill-buffer-hook= とアイドルタイマーで、
本モジュールは =buffer-list-update-hook= で、同じことをしていました。どちらも冪等で害はありませんでしたが、
仕事が重複していたため、2026-10-18 に本モジュールへ統合し、 =core-buffers= を廃止しました。

機能は 2 つです。

1. *自動再生成* — =scratch--ensure-exists= を =buffer-list-update-hook= に登録しています。
   このフックはバッファリストが変わるたびに発火するので、 =*scratch*= を kill しても、後続の
   コマンドがその不在に気づく前に =lisp-interaction-mode= で作り直されます。
   =unless (get-buffer "*scratch*")= のガードがあるため、通常時のホットパスはハッシュ参照
   1 回で済みます。

2. *複数 scratch* — =my/scratch-new= が 2 から順に空いている名前を探し、 =*scratch-N*=
   （N = 2, 3, …）を作ります。各バッファは =lisp-interaction-mode= で初期化され、内容として
   =scratch-initial-message= が挿入されます。

なぜ =kill-buffer-hook= ではないのか

=kill-buffer-hook= は、kill されようとしているバッファのコンテキストの中で走ります。そこから
同名のバッファを作ろうとすると、順序上の危険が生じます。 =buffer-list-update-hook= なら kill が
完全に終わった後に走るので、この問題を避けられます。

*コーディング規則との対応*

- フック関数 =scratch--ensure-exists= はトップレベルの名前付き =defun= です
  （トップレベルの名前付き defun にする — 規則 8）
- =scratch-initial-message= は =defcustom= で、その =defgroup= は本モジュールが宣言しています
  （規則 9）
- =defcustom= への代入は =setopt= 、 =defvar= への代入は =setq= です（規則 6）

#+begin_src emacs-lisp
  ;;; utils/utils-scratch.el --- scratch バッファの管理 -*- lexical-binding: t -*-

  ;; Copyright (C) 2024  YAMASHITA, Takao
  ;; Author: YAMASHITA, Takao <tjy1965@gmail.com>

  ;;; Commentary:

  ;; 提供する機能は 2 つ:
  ;;
  ;; 1. 自動再生成: `scratch--ensure-exists' を `buffer-list-update-hook' に登録する。
  ;;    *scratch* が kill されると、後続のコマンドに制御が戻る前に
  ;;    `lisp-interaction-mode' で再生成される。
  ;;
  ;;
  ;; 2. 複数 scratch: `my/scratch-new' が *scratch-N*（N >= 2）を作成し、そこへ切り替える。
  ;;

  ;;; Code:

  (defgroup scratch nil
    "Scratch buffer management."
    :group 'convenience
    :prefix "scratch-")

  (defcustom scratch-initial-message
    ";; This buffer is for text that is not saved.\n\n"
    "Initial content inserted into a newly created scratch buffer."
    :type 'string
    :group 'scratch)

  ;;; 内部ヘルパー

  (defun scratch--ensure-exists ()
    "Recreate *scratch* if it does not currently exist.
  Registered on `buffer-list-update-hook'.  The guard keeps the common
  case to a single hash lookup so repeated hook invocations are cheap."
    (unless (get-buffer "*scratch*")
      (with-current-buffer (get-buffer-create "*scratch*")
        (lisp-interaction-mode)
        (when (zerop (buffer-size))
          (insert scratch-initial-message)))))

  ;;; 公開コマンド

  (defun my/scratch-new ()
    "Create and switch to a new scratch buffer named *scratch-N*.
  N is the smallest integer >= 2 for which the name is not already in
  use.  The buffer is initialised in `lisp-interaction-mode'."
    (interactive)
    (let* ((n 2)
           (name (format "*scratch-%d*" n)))
      (while (get-buffer name)
        (setq n (1+ n))
        (setq name (format "*scratch-%d*" n)))
      (switch-to-buffer (get-buffer-create name))
      (lisp-interaction-mode)
      (when (zerop (buffer-size))
        (insert scratch-initial-message))
      (message "Created %s" name)))

  ;;; セットアップ

  (defun scratch--setup ()
    "Register the scratch auto-recreation hook."
    (add-hook 'buffer-list-update-hook #'scratch--ensure-exists))

  (scratch--setup)

  (provide 'utils-scratch)
  ;;; utils/utils-scratch.el ends here
#+end_src

#+begin_src emacs-lisp
  ;;; utils/utils-claude.el --- Claude API の HTTP クライアント -*- lexical-binding: t; -*-
  ;;
  ;; Copyright (c) 2021-2026
  ;; Author: YAMASHITA, Takao
  ;; License: GNU GPL v3 or later
  ;;
  ;; Category: utils
  ;;
  ;;; Commentary:
  ;;
  ;; Anthropic Claude Messages API のための低レベル HTTP クライアント。
  ;; 本モジュールは UI、Org、および utils より上位のいかなる層にも依存しない。
  ;;
  ;; 公開 API:
  ;;   `utils-claude-api-key-function' — API キーを返す関数（personal 層が設定する）
  ;;   `utils-claude-get-api-key'  — 上の関数でキーを取得する。無ければ nil
  ;;   `utils-claude-request-sync' — 同期 POST。応答テキストを返す
  ;;   `utils-claude-request'      — コールバックを伴う非同期 POST
  ;;
  ;; API キーの取得先（auth-source など）を、本モジュールは知らない。取得は personal 層の
  ;; `personal-credentials' が担い、`utils-claude-api-key-function' へ渡す。下位層である
  ;; utils は上位層を参照できないため、関数を差し込む形で依存の向きを保っている。
  ;; 本モジュールは `personal-ai' が `require' して使う（`personal-ai' は user.el が Org の
  ;; ロード後に読み込む）。
  ;;
  ;; 既定のモデルは `utils-claude-default-model' で変更できる。
  ;; 既定の max-tokens は `utils-claude-max-tokens' で変更できる。
  ;;
  ;; エラー処理:
  ;;   同期版は、HTTP レベルの失敗（200 以外のステータス）、JSON のパースエラー、
  ;;   API キーの欠落で `utils-claude-error' をシグナルする。呼び出し側は
  ;;   `condition-case' で包むこと。非同期版はシグナルせず、同じ失敗を
  ;;   "ERROR:" で始まる文字列としてコールバックへ渡す。
  ;;
  ;;; Code:

  (eval-when-compile
    (require 'cl-lib)
    (require 'subr-x))

  (require 'url)
  (require 'url-http)
  (require 'json)

  (defgroup utils-claude nil
    "Claude API HTTP client for Emacs."
    :group 'tools
    :prefix "utils-claude-")

  (defcustom utils-claude-default-model "claude-sonnet-4-5"
    "Default Anthropic model identifier used when no model is specified."
    :type 'string
    :group 'utils-claude)

  (defcustom utils-claude-max-tokens 4096
    "Default maximum tokens in the completion response."
    :type 'natnum
    :group 'utils-claude)

  (defcustom utils-claude-api-version "2023-06-01"
    "Anthropic API version header value."
    :type 'string
    :group 'utils-claude)

  (defcustom utils-claude-endpoint "https://api.anthropic.com/v1/messages"
    "Endpoint URL for the Anthropic Messages API."
    :type 'string
    :group 'utils-claude)

  (define-error 'utils-claude-error "Claude API error" 'error)

  (defcustom utils-claude-api-key-function nil
    "Function of no arguments that returns the Anthropic API key, or nil.
  The personal layer sets it (`personal-credentials-anthropic-key'), so
  this lower layer never learns where the key is stored."
    :type '(choice (const :tag "None" nil) function)
    :group 'utils-claude)

  (defun utils-claude-get-api-key ()
    "Return the Anthropic API key from `utils-claude-api-key-function', or nil."
    (when (functionp utils-claude-api-key-function)
      (funcall utils-claude-api-key-function)))

  (defun utils-claude--build-headers (api-key)
    "Return the alist of HTTP headers required by the Anthropic API.
  API-KEY is the raw secret string."
    `(("Content-Type"      . "application/json")
      ("x-api-key"         . ,api-key)
      ("anthropic-version" . ,utils-claude-api-version)))

  (defun utils-claude--build-body (messages &optional model max-tokens system)
    "Return a JSON string representing the Messages API request body.
  MESSAGES is a list of plists with :role and :content keys.
  MODEL defaults to `utils-claude-default-model'.
  MAX-TOKENS defaults to `utils-claude-max-tokens'.
  SYSTEM, when non-nil, is inserted as the system prompt string."
    (let ((payload
           `((model      . ,(or model utils-claude-default-model))
             (max_tokens . ,(or max-tokens utils-claude-max-tokens))
             (messages   . ,(apply #'vector
                                   (mapcar (lambda (m)
                                             `((role    . ,(plist-get m :role))
                                               (content . ,(plist-get m :content))))
                                           messages))))))
      (when system
        (push `(system . ,system) payload))
      (encode-coding-string (json-encode payload) 'utf-8)))

  (defun utils-claude--parse-response (buffer)
    "Extract the assistant text content from the HTTP response in BUFFER.
  Returns the text string on success.
  Signals `utils-claude-error' on HTTP error or unexpected JSON shape."
    (with-current-buffer buffer
      (goto-char (point-min))
      (let ((status (url-http-parse-response)))
        (unless (= status 200)
          (signal 'utils-claude-error
                  (list (format "HTTP %d from Anthropic API" status)))))
      (re-search-forward "\r?\n\r?\n" nil t)
      (let* ((json-object-type 'plist)
             (json-array-type  'list)
             (json-key-type    'keyword)
             (data (condition-case err
                       (json-read)
                     (json-error
                      (signal 'utils-claude-error
                              (list (format "JSON parse error: %s"
                                            (error-message-string err)))))))
             (content (plist-get data :content)))
        (or (plist-get (car content) :text)
            (signal 'utils-claude-error
                    (list "Unexpected response shape: no :content[0]:text"))))))

  (defun utils-claude-request-sync (messages &optional model max-tokens system)
    "Send MESSAGES to the Claude API synchronously and return the response text.
  MESSAGES is a list of plists with :role and :content string values.
  MODEL, MAX-TOKENS, and SYSTEM are forwarded to `utils-claude--build-body'.
  Signals `utils-claude-error' on failure."
    (let* ((api-key (or (utils-claude-get-api-key)
                        (signal 'utils-claude-error
                                '("No API key: `utils-claude-api-key-function' returned nil"))))
           (url-request-method "POST")
           (url-request-extra-headers (utils-claude--build-headers api-key))
           (url-request-data (utils-claude--build-body
                              messages model max-tokens system))
           (response-buffer (url-retrieve-synchronously
                             utils-claude-endpoint t nil 30)))
      (unwind-protect
          (utils-claude--parse-response response-buffer)
        (when (buffer-live-p response-buffer)
          (kill-buffer response-buffer)))))

  (defun utils-claude-request (messages callback &optional model max-tokens system)
    "Send MESSAGES to the Claude API asynchronously and call CALLBACK.
  CALLBACK receives the response text string on success, or a string
  beginning with \"ERROR:\" on failure.
  MODEL, MAX-TOKENS, and SYSTEM are forwarded to `utils-claude--build-body'."
    (let* ((api-key (utils-claude-get-api-key))
           (url-request-method "POST")
           (url-request-extra-headers
            (if api-key
                (utils-claude--build-headers api-key)
              (funcall callback "ERROR: No API key (see `utils-claude-api-key-function')")
              nil))
           (url-request-data
            (when url-request-extra-headers
              (utils-claude--build-body messages model max-tokens system))))
      (when url-request-extra-headers
        (url-retrieve
         utils-claude-endpoint
         (lambda (status)
           (if (plist-get status :error)
               (funcall callback
                        (format "ERROR: %s"
                                (error-message-string (plist-get status :error))))
             (condition-case err
                 (funcall callback (utils-claude--parse-response (current-buffer)))
               (utils-claude-error
                (funcall callback (format "ERROR: %s" (cadr err)))))))
         nil t))))
