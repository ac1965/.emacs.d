#+begin_src emacs-lisp
  ;;; dev/dev-lsp-bridge.el --- LSP 抽象化の補助 -*- lexical-binding: t; -*-
  ;;
  ;;
  ;; Copyright (c) 2021-2026
  ;; Author: YAMASHITA, Takao
  ;; License: GNU GPL v3 or later
  ;;
  ;; Category: dev
  ;;
  ;;; Commentary:
  ;;
  ;; LSP 抽象化層: バックエンド非依存の code-actions、rename、format。
  ;;
  ;; 本ファイルの責務:
  ;; - `dev-lsp-code-actions'、`dev-lsp-rename'、`dev-lsp-format-buffer' を
  ;;   統一された対話コマンドとして提供する
  ;; - 呼び出し時に `cond'/`fboundp' で eglot か lsp-mode にディスパッチする
  ;; - LSP が動いていない場合、整形は `indent-region' にフォールバックする
  ;; - 上記 3 コマンドを `ui-leader-c-map' （M-SPC c a / r / f）へ自分で割り当てる
  ;;   （`dev-lsp-bridge--bind-keys'）
  ;;
  ;; キー割り当てに関する注記:
  ;; ui-keys.el がこの 3 コマンドを直接参照すると、ui 層が dev 層に依存することになり、
  ;; 層の方向（下位が上位に依存しない）に反する。そのため割り当ての責務は本モジュールが
  ;; 持つ。マップ（`ui-leader-c-map'）は ui-keymap の
  ;; `after-init-hook' で作られる（ui-leader--define-maps）ので、本モジュールの hook は
  ;; depth 90 で登録し、必ずその後に走らせる。マップが無ければ何もしない。
  ;;
  ;; ロード順に関する注記:
  ;; dev/ セクションの先頭、具体的な LSP バックエンドのモジュール（dev-lsp-eglot、
  ;; dev-lsp-mode）より前にロードされる。これらのディスパッチ・ラッパー自体は
  ;; バックエンドに依存しないため、dev/ に置くのがアーキテクチャ上正しい。
  ;; personal オーバーレイも `dev-lsp-*' コマンドを特定のバックエンドに依存せず
  ;; バインドできる。具体的なバックエンドは core-switches が
  ;; `after-init-hook' で遅延的に有効化する。
  ;;; Code:

  ;; バイトコンパイル時の警告を避けるための宣言（実体は各パッケージ／ライブラリ側にある）。
  (declare-function eglot-code-actions "eglot")
  (declare-function eglot-format-buffer "eglot")
  (declare-function eglot-rename "eglot")
  (declare-function lsp-execute-code-action "lsp-mode")
  (declare-function lsp-format-buffer "lsp-mode")
  (declare-function lsp-rename "lsp-mode")

  ;;;###autoload
  (defun dev-lsp--dispatch (eglot-fn lsp-fn fallback-fn error-msg)
    "Dispatch to the active LSP backend using EGLOT-FN, LSP-FN, or FALLBACK-FN.
  Tries each in order via `fboundp'; calls it interactively if bound.
  Signals `user-error' with ERROR-MSG when none is available.
  FALLBACK-FN may be nil to skip the fallback clause."
    (cond
     ((fboundp eglot-fn)   (call-interactively eglot-fn))
     ((fboundp lsp-fn)     (call-interactively lsp-fn))
     ((and fallback-fn
           (fboundp fallback-fn)) (call-interactively fallback-fn))
     (t (user-error "%s" error-msg))))

  ;;;###autoload
  (defun dev-lsp-code-actions ()
    "Run code actions at point via the active LSP backend (eglot or lsp-mode)."
    (interactive)
    (dev-lsp--dispatch #'eglot-code-actions #'lsp-execute-code-action nil
                       "No LSP backend available"))

  ;;;###autoload
  (defun dev-lsp-rename ()
    "Rename symbol at point via the active LSP backend (eglot or lsp-mode)."
    (interactive)
    (dev-lsp--dispatch #'eglot-rename #'lsp-rename nil
                       "No LSP backend available"))

  ;;;###autoload
  (defun dev-lsp-format-buffer ()
    "Format the current buffer via the LSP backend, else `indent-region'."
    (interactive)
    (dev-lsp--dispatch #'eglot-format-buffer #'lsp-format-buffer #'indent-region
                       "No formatter available"))

  (defvar ui-leader-c-map)

  (defun dev-lsp-bridge--bind-keys ()
    "Bind the LSP commands in `ui-leader-c-map' when the map exists."
    (when (and (boundp 'ui-leader-c-map) (keymapp ui-leader-c-map))
      (keymap-set ui-leader-c-map "a" #'dev-lsp-code-actions)
      (keymap-set ui-leader-c-map "r" #'dev-lsp-rename)
      (keymap-set ui-leader-c-map "f" #'dev-lsp-format-buffer)))

  ;; depth 90: ui-keymap--startup-init （depth 0）がマップを作った後に走らせる。
  (add-hook 'after-init-hook #'dev-lsp-bridge--bind-keys 90)

  ;; LSP バックエンドを `core-switches' の登録簿へ登録する。core 層の core-switches は
  ;; バックエンドのモジュール名を知らない。名前を知る本モジュールが登録する。
  (require 'core-switches)
  (core-switches-register 'lsp 'eglot 'my/lsp-enable-eglot
                          "dev/dev-lsp-eglot" "Eglot setup" "eglot")
  (core-switches-register 'lsp 'lsp 'my/lsp-enable-lspmode
                          "dev/dev-lsp-mode" "lsp-mode setup" "lsp-mode")

  (provide 'dev-lsp-bridge)
  ;;; dev/dev-lsp-bridge.el ends here
#+end_src

#+begin_src emacs-lisp
  ;;; dev/dev-lsp-eglot.el --- Eglot による LSP 設定 -*- lexical-binding: t; -*-
  ;;
  ;;
  ;; Copyright (c) 2021-2026
  ;; Author: YAMASHITA, Takao
  ;; License: GNU GPL v3 or later
  ;;
  ;; Category: dev
  ;;
  ;;; Commentary:
  ;;
  ;; プロジェクト推測可否のガードを備えた Eglot の LSP 設定。
  ;;
  ;; 本ファイルの責務:
  ;; - `eglot-ensure' を呼ぶ前に、現在のバッファに対して eglot がサーバを見つけられるか
  ;;   を判定する `my/eglot-guessable-p' を定義する
  ;; - `dev-lsp-eglot--maybe-ensure' を `prog-mode-hook' にフックする。ファイルを訪問する
  ;;   バッファで初めて呼ばれたときに eglot をロードする（ロードしないと `eglot--guess-contact'
  ;;   が無く、判定が常に nil になって eglot が一度も起動しない。*scratch* では呼ばない）
  ;; - C/C++ は clangd（--background-index --clang-tidy）、Python は pyright を
  ;;   `eglot-server-programs' に登録する（clangd は Homebrew の llvm を優先）
  ;; - 最後のバッファを閉じたらサーバを止める（`eglot-autoshutdown'）
  ;; - サーバが動的に登録するファイル監視（`workspace/didChangeWatchedFiles'）を受け付けない
  ;;   （`dev-lsp-eglot--skip-file-watchers'）。理由は下の「ファイル監視を受け付けない理由」
  ;; - eglot は Emacs 同梱を使う（`:straight nil'。コーディング規則 3）。`:straight t' だと
  ;;   ELPA から eglot と、その依存の eldoc・flymake・project・xref・jsonrpc などまで
  ;;   取得して同梱版を上書きし、初回の起動も遅くなる
  ;; - core-switches 向けのエントリポイント `my/lsp-enable-eglot' を提供する
  ;;
  ;; rename / code-actions は `dev-lsp-rename'、`dev-lsp-code-actions'（ui-keymap の
  ;; leader-c-map）で呼ぶ。`C-c r' は query-replace に使っているため eglot-mode-map には
  ;; 割り当てない。
  ;;
  ;; ファイル監視を受け付けない理由:
  ;; pyright は起動直後に、Python のインストール先（site-packages）全体を対象にした
  ;; ファイル監視を登録してくる。Eglot はこれを kqueue で監視するが、kqueue は
  ;; ファイル 1 つにつきファイル記述子を 1 つ使うため、site-packages のような大きな木では
  ;; 記述子が尽きて `file-notify-error' になる。Eglot はサーバへ「Internal error」と返し、
  ;; pyright はそれを受けて終了コード 1 で落ちる。Eglot は再接続を繰り返し、補完要求は
  ;; 「Server died」で失敗する（2026-10-31 に M-x my/smoketest-lsp で見つかった）。
  ;; pylsp でも同じ形で落ちた。代わりに失うのは、エディタの外でファイルが変わったときの
  ;; 通知だけである。サーバは開いているバッファの内容を didChange で受け取るので、
  ;; 編集中の補完・診断・定義ジャンプには影響しない。
  ;;; Code:

  (eval-when-compile (require 'leaf))

  ;; バイトコンパイル時の警告を避けるための宣言（実体は eglot 側にある）。
  (defvar eglot-server-programs)

  (defun dev-lsp-eglot--skip-file-watchers (orig server method &rest args)
    "Around advice for `eglot-register-capability': ignore file-watch registrations.
  Call ORIG with SERVER, METHOD and ARGS for every other method.  Return nil
  without calling ORIG when METHOD is `workspace/didChangeWatchedFiles'."
    (unless (eq method 'workspace/didChangeWatchedFiles)
      (apply orig server method args)))

  (defun my/eglot-guessable-p ()
    "Return non-nil if Eglot can guess a server for the current buffer.
  `eglot--guess-contact' returns (MODES PROJECT CLASS CONTACT LANGUAGE-IDS) even
  when no server program is registered for the mode (CONTACT is then nil, as in
  `emacs-lisp-mode'), so a non-nil result alone is not enough: connecting with a
  nil CONTACT fails with \"Wrong type argument: processp, nil\"."
    (let ((guess (cond ((fboundp 'eglot--guess-contact)
                        (ignore-errors (eglot--guess-contact)))
                       ((fboundp 'eglot-guess-contact)
                        (ignore-errors (eglot-guess-contact))))))
      (and guess (nth 3 guess))))

  ;;;###autoload
  (defun dev-lsp-eglot--maybe-ensure ()
    "Enable eglot in the current file buffer when a server can be guessed.
  Eglot is loaded here on first use: `my/eglot-guessable-p' needs
  `eglot--guess-contact', which does not exist until eglot is loaded, so without
  this `require' the guard is always nil and eglot is never started.  Buffers
  that visit no file (*scratch* and the like) are skipped, so eglot (about
  130 ms to load) is not pulled in at startup."
    (when (and buffer-file-name
               (require 'eglot nil t)
               (my/eglot-guessable-p))
      (eglot-ensure)))

  ;; defun は leaf より前に置く（vcs-core.el の smerge と同じ理由。:hook に関数名を
  ;; 書くと leaf が autoload と declare-function を出し、後ろだと二重定義の警告になる）。
  (leaf eglot
    :straight nil
    :commands (eglot eglot-ensure)
    :custom ((eglot-autoreconnect . t)
             (eglot-autoshutdown . t))
    :hook ((prog-mode . dev-lsp-eglot--maybe-ensure))
    :config
    (advice-add 'eglot-register-capability :around #'dev-lsp-eglot--skip-file-watchers)
    (add-to-list 'eglot-server-programs
                 `((c-mode c-ts-mode c++-mode c++-ts-mode)
                   . (,(let ((brew "/opt/homebrew/opt/llvm/bin/clangd"))
                         (if (file-executable-p brew) brew "clangd"))
                      "--background-index" "--clang-tidy")))
    (add-to-list 'eglot-server-programs
                 '((python-mode python-ts-mode)
                   . ("pyright-langserver" "--stdio"))))

  ;;;###autoload
  (defun my/lsp-enable-eglot ()
    "Enable the Eglot LSP backend for all future `prog-mode' buffers."
    (interactive)
    (add-hook 'prog-mode-hook #'dev-lsp-eglot--maybe-ensure)
    (message "[lsp] Eglot enabled."))

  (provide 'dev-lsp-eglot)
  ;;; dev-lsp-eglot.el ends here
#+end_src

* dev/dev-ai.el
:PROPERTIES:
:CUSTOM_ID: dev-ai
:header-args:emacs-lisp: :tangle ../lisp/dev/dev-ai.el
:END:

AI 支援開発の口です。 =agent-shell= （Claude Code などのエージェントの Emacs 画面。ACP）を導入し、
認証の方法を選べるようにします。

*認証の選び方*

| =dev-ai-agent-authentication= | 認証 | API キー |
|---+---+---|
| =login= （既定） | =claude= のログイン（Pro / Max の契約） | 要らない |
| =api-key= | personal 層が渡す Anthropic の API キー | =dev-ai-anthropic-key-function= が返す |

=claude-agent-acp= は、自分では認証しません。既定の =login= では、 =claude= が保存したログインを使うので、
Pro / Max の契約の枠で動きます。 =api-key= にすると、 =claude-agent-acp= も API キーで動きます（課金は API の従量課金）。

API キーの出どころ（ =auth-source= ）は、dev 層は知りません。personal 層の =personal-credentials= が
関数として渡します。

#+begin_src emacs-lisp
  ;;; dev/dev-ai.el --- AI 支援開発の補助 -*- lexical-binding: t; -*-
  ;;
  ;; Copyright (c) 2021-2026
  ;; Author: YAMASHITA, Takao
  ;; License: GNU GPL v3 or later
  ;;
  ;; Category: dev
  ;;
  ;;; Commentary:
  ;;
  ;; agent-shell（Claude Code などのエージェントの Emacs 画面。ACP）による AI 支援開発。
  ;;
  ;; 本ファイルの責務:
  ;; - agent-shell を導入し、M-SPC a c（Claude Code を起動）と M-SPC a C（エージェントを選んで
  ;;   起動）に割り当てる（`dev-ai--bind-keys'）。M-SPC a c は、ACP クライアント
  ;;   （`claude-agent-acp'）の実行ファイルがあるときだけ割り当てる
  ;; - 認証の方法を `dev-ai-agent-authentication' で選べるようにする。既定は `login'
  ;;   （`claude' のログイン。API キーは要らない）。`api-key' にすると、
  ;;   `dev-ai-anthropic-key-function' が返すキーを使う。関数が無ければ警告して `login' に落ちる
  ;;
  ;; API キーの取得先（auth-source など）を、本モジュールは知らない。取得は personal 層の
  ;; `personal-credentials' が担い、`dev-ai-anthropic-key-function' へ渡す。下位層である dev は
  ;; 上位層の personal を参照できないため、関数を差し込む形で依存の向きを保っている。
  ;;
  ;;; Code:

  (eval-when-compile (require 'leaf))

  ;; バイトコンパイル時の警告を避けるための宣言（実体は各パッケージ側にある）。
  (defvar ui-leader-a-map)
  (defvar agent-shell-anthropic-authentication)
  (declare-function agent-shell-anthropic-make-authentication "agent-shell-anthropic")
  (declare-function agent-shell-anthropic-start-claude-code "agent-shell-anthropic")
  (declare-function agent-shell "agent-shell")

  (defgroup dev-ai nil
    "AI-assisted development helpers."
    :group 'tools)

  (defcustom dev-ai-agent-authentication 'login
    "How agent-shell authenticates to Claude Code.
  `login' uses the logged-in `claude' CLI (a Pro or Max subscription); no API
  key is needed.  `api-key' uses the key `dev-ai-anthropic-key-function' returns."
    :type '(choice (const :tag "claude CLI login" login)
                   (const :tag "Anthropic API key" api-key))
    :group 'dev-ai)

  (defcustom dev-ai-anthropic-key-function nil
    "Function of no arguments that returns the Anthropic API key, or nil.
  Only used when `dev-ai-agent-authentication' is `api-key'.  The personal
  layer sets it (`personal-credentials-anthropic-key'); this layer never
  learns where the key is stored."
    :type '(choice (const :tag "None" nil) function)
    :group 'dev-ai)

  (defun dev-ai--agent-authentication ()
    "Return the agent-shell authentication for `dev-ai-agent-authentication'.
  When `api-key' is chosen but `dev-ai-anthropic-key-function' is not set,
  warn and fall back to the `claude' login instead of failing at start-up."
    (when (and (eq dev-ai-agent-authentication 'api-key)
               (not (functionp dev-ai-anthropic-key-function)))
      (display-warning 'dev-ai "api-key が選ばれたが dev-ai-anthropic-key-function が無い。claude のログインで認証する"))
    (if (and (eq dev-ai-agent-authentication 'api-key)
             (functionp dev-ai-anthropic-key-function))
        (agent-shell-anthropic-make-authentication
         :api-key (lambda () (funcall dev-ai-anthropic-key-function)))
      (agent-shell-anthropic-make-authentication :login t)))

  (defun dev-ai--acp-available-p ()
    "Return non-nil when the Claude Code ACP client executable is on `exec-path'."
    (executable-find (if (boundp 'agent-shell-anthropic-claude-acp-command)
                         (car agent-shell-anthropic-claude-acp-command)
                       "claude-agent-acp")))

  (defun dev-ai--configure-agent-shell ()
    "Apply `dev-ai-agent-authentication' to agent-shell's Claude Code client."
    (setopt agent-shell-anthropic-authentication (dev-ai--agent-authentication)))

  (defun dev-ai--bind-keys ()
    "Bind the AI commands in `ui-leader-a-map' when it exists.
  The Claude Code key is bound only when the ACP client executable exists."
    (when (and (boundp 'ui-leader-a-map) (keymapp ui-leader-a-map))
      (when (dev-ai--acp-available-p)
        (keymap-set ui-leader-a-map "c" #'agent-shell-anthropic-start-claude-code))
      (keymap-set ui-leader-a-map "C" #'agent-shell)))

  ;; depth 90: ui-keymap--startup-init（depth 0）がマップを作った後に走らせる。
  (add-hook 'after-init-hook #'dev-ai--bind-keys 90)

  (leaf agent-shell
    :straight t
    :commands (agent-shell agent-shell-anthropic-start-claude-code)
    :config (dev-ai--configure-agent-shell))

  (provide 'dev-ai)
  ;;; dev/dev-ai.el ends here
#+end_src

  ;;; dev/dev-term.el --- ターミナル統合の補助 -*- lexical-binding: t; -*-
  ;;
  ;; Copyright (c) 2021-2026
  ;; Author: YAMASHITA, Takao
  ;; License: GNU GPL v3 or later
  ;;
  ;; Category: dev
  ;;
  ;;; Commentary:
  ;;
  ;; vterm を基盤とするターミナル統合の補助。
  ;;
  ;; 提供するもの:
  ;;
  ;; - ターミナル色への Nano パレット統合（任意。ui-nano-palette）
  ;; - vterm-toggle によるプロジェクト単位のターミナル切替
  ;; - 画面下部にドッキングするターミナルウィンドウ表示

     (emms-info-functions                . '(emms-info-native))
     (emms-player-list                   . '(emms-player-mpv emms-player-mplayer)))

    (require 'emms-setup)
    (emms-all)
    (emms-default-players)

    (setopt emms-history-file (expand-file-name "history" emms-directory))

    (my/ensure-directory-exists emms-directory)
    (emms-mode-line-mode 1))

    :straight nil
    :after emms)

  (when dev-music-enable-p

  ;; AppleScript による Apple Music 操作
  ;;
  ;; apple-music--osascript 内の darwin ガードにより、macOS 以外では全コマンドが
  ;; 何もしない。osascript はシェルを介さず call-process で呼ぶ。状態を読む操作は
  ;; Music.app が未起動なら何もせず、起動するのは play、トグル、プレイリスト再生、
  ;; 検索再生、アルバムからの再生だけである。曲順一覧（t）は、再生元がライブラリなど
  ;; 組み込みリストなら現在の曲のアルバム、それ以外ならそのプレイリストを曲順に並べる。
  ;; hydra は after-init で定義する。これにより hydra のロードが保証される
  ;; （hydra は ui-keys.el が要求する straight.el のパッケージであり、ui-keys.el は
  ;; after-init-hook の発火より前に my:modules 経由で走る）。

  (defun apple-music--osascript (lines)
    "Run osascript with the script LINES (a list of strings), one -e each.
    Return the trimmed output, or nil on non-macOS systems or when osascript
    exits with an error.  No shell is involved, so LINES need no shell quoting."
    (when (eq system-type 'darwin)
      (with-temp-buffer
        (when (zerop (apply #'call-process "osascript" nil t nil
                            (mapcan (lambda (l) (list "-e" l)) lines)))
          (string-trim (buffer-string))))))

  (defun apple-music--quote (string)
    "Return STRING escaped for use inside an AppleScript string literal."
    (replace-regexp-in-string "[\\\"]" "\\\\\\&" string))

  (defun apple-music--applescript (cmd &optional launch)
    "Send CMD to Music.app via osascript.  No-op on non-macOS systems.
    Unless LAUNCH is non-nil, do nothing when Music.app is not running, so a
    query or a pause never starts the application."
    (let ((line (format "tell application \"Music\" to %s" cmd)))
      (apple-music--osascript
       (if launch
           (list line)
         (list "if application \"Music\" is running then" line "end if")))))

    The names are joined with linefeeds on the AppleScript side, so a name that
    contains a comma stays in one piece.  Returns nil on non-macOS systems or
    when Music.app is not running."
    (let ((raw (apple-music--osascript
                (list "if application \"Music\" is running then"
                      "tell application \"Music\""

    "Return a plist describing the current track, or nil.
    The keys are :name :artist :album :state :position and :duration (the last

    current track."
    (let ((raw (apple-music--osascript
                (list "if application \"Music\" is running then"
                      "tell application \"Music\""

                      "set pp to player position"
                      "set dur to duration of t"
                      (concat "return (name of t) & linefeed & (artist of t) & linefeed"
                              " & (album of t) & linefeed & (player state as text)"

                      "end tell"
                      "end if"))))

  (defun apple-music--search (query)
    "Return songs matching QUERY as a list of (NAME . ARTIST), best first.
    At most `apple-music-search-limit' songs are returned.  Return nil when
    nothing matches or Music.app is not running."
    (let ((raw (apple-music--osascript
                (list "if application \"Music\" is running then"
                      "tell application \"Music\""
                      (format "set rs to search (library playlist 1) for \"%s\" only songs"
                              (apple-music--quote query))
                      "set n to count of rs"
                      (format "if n > %d then set n to %d"
                              apple-music-search-limit apple-music-search-limit)
                      "set out to {}"
                      "repeat with i from 1 to n"
                      "set t to item i of rs"
                      "set end of out to (name of t) & tab & (artist of t)"
                      "end repeat"
                      "set AppleScript's text item delimiters to linefeed"
                      "return out as text"
                      "end tell"
                      "end if"))))
      (when (and raw (not (string-empty-p raw)))
        (mapcar (lambda (line)
                  (let ((p (split-string line "\t")))
                    (cons (car p) (or (cadr p) ""))))
                (split-string raw "\n" t)))))

  (defun apple-music-search-play (query)
    "Search the Music.app library for QUERY and play the song you choose."
    (interactive (list (read-string "Search Music: ")))
    (when (string-empty-p query)
      (user-error "Empty query"))
    (let ((results (apple-music--search query)))
      (if (null results)
          (message "[apple-music] No songs found for \"%s\"" query)
        (let* ((i 0)
               (cands (mapcar (lambda (r)
                                (format "%d. %s — %s" (setq i (1+ i)) (car r) (cdr r)))
                              results))
               (choice (completing-read "Song: " cands nil t))
               (idx (string-to-number choice)))
          (when (> idx 0)
            (apple-music--applescript
             (format "play (item %d of (search (library playlist 1) for \"%s\" only songs))"
                     idx (apple-music--quote query))
             t)
            (message "♫ Playing: %s" (substring choice (1+ (string-search " " choice)))))))))

  (defun apple-music--choose-track (prompt rows current)
    "Prompt with PROMPT to pick one of ROWS and return its 1-based index.
    ROWS is a list of (NAME . ARTIST) in play order.  CURRENT is the 1-based
    index of the playing track (0 if unknown); it is marked and preselected.
    The candidates keep ROWS order instead of being sorted."
    (let* ((i 0)
           (cands (mapcar (lambda (r)
                            (setq i (1+ i))
                            (format "%d.%s %s — %s" i (if (= i current) " ▶" "")
                                    (car r) (cdr r)))
                          rows))
           (table (lambda (str pred action)
                    (if (eq action 'metadata)
                        '(metadata (display-sort-function . identity)
                                   (cycle-sort-function . identity))
                      (complete-with-action action cands str pred))))
           (choice (completing-read prompt table nil t nil nil
                                    (and (> current 0) (nth (1- current) cands))))
           (idx (string-to-number choice)))
      (and (> idx 0) idx)))

  (defun apple-music--playlist-tracks ()
    "Return (PLAYLIST-NAME CURRENT . ROWS) for the playlist now playing, or nil.
    ROWS is a list of (NAME . ARTIST) in play order and CURRENT is the 1-based
    index of the current track in it (0 if unknown)."
    (let ((raw (apple-music--osascript
                (list "if application \"Music\" is running then"
                      "tell application \"Music\""
                      "set pl to current playlist"
                      "set ns to name of every track of pl"
                      "set ars to artist of every track of pl"
                      "set cur to 0"
                      "try"
                      "set cur to index of current track"
                      "end try"
                      "set out to {(name of pl) & tab & (cur as text)}"
                      "repeat with i from 1 to count of ns"
                      "set end of out to (item i of ns) & tab & (item i of ars)"
                      "end repeat"
                      "set AppleScript's text item delimiters to linefeed"
                      "return out as text"
                      "end tell"
                      "end if"))))
      (when (and raw (not (string-empty-p raw)))
        (let* ((lines (split-string raw "\n"))
               (head (split-string (car lines) "\t")))
          (when (cdr lines)
            (cons (car head)
                  (cons (string-to-number (or (cadr head) "0"))
                        (mapcar (lambda (line)
                                  (let ((p (split-string line "\t")))
                                    (cons (car p) (or (cadr p) ""))))
                                (cdr lines)))))))))

  (defun apple-music-track-select ()
    "Choose a track from what is now playing and play it.
    When a user playlist or an album playlist is playing, list its tracks in
    play order.  When playback comes from a built-in list such as the Library
    (`special kind' is not none), list the album of the current track instead.
    The current track is marked and preselected."
    (interactive)
    (let ((builtin (apple-music--osascript
                    (list "if application \"Music\" is running then"
                          "tell application \"Music\""
                          "set pl to current playlist"
                          "if name of pl is name of library playlist 1 then return \"true\""
                          "return ((special kind of pl) is not none) as text"
                          "end tell"
                          "end if")))
          (data nil))
      (cond
       ((equal builtin "true")
        (apple-music-album-select))
       ((null (setq data (apple-music--playlist-tracks)))
        (message "[apple-music] No current playlist (is Music.app playing?)"))
       (t
        (let ((idx (apple-music--choose-track
                    (format "Track [%s]: " (car data)) (cddr data) (cadr data))))
          (when idx
            (apple-music--applescript
             (format "play (track %d of current playlist)" idx))
            (message "♫ Playing: %s" (car (nth (1- idx) (cddr data))))))))))

  (defun apple-music--album-tracks ()
    "Return (ALBUM-NAME . ROWS) for the album of the current track, or nil.
    ROWS is a list of (NAME ARTIST ID), sorted by disc and track number; ID is
    the persistent ID used to play the track."
    (let ((raw (apple-music--osascript
                (list "if application \"Music\" is running then"
                      "tell application \"Music\""
                      "set al to album of current track"
                      "set ts to (every track of library playlist 1 whose album is al)"
                      "set out to {al}"
                      "repeat with t in ts"
                      (concat "set end of out to ((disc number of t) as text) & tab"
                              " & ((track number of t) as text) & tab & (name of t)"
                              " & tab & (artist of t) & tab & (persistent ID of t)")
                      "end repeat"
                      "set AppleScript's text item delimiters to linefeed"
                      "return out as text"
                      "end tell"
                      "end if"))))
      (when (and raw (not (string-empty-p raw)))
        (let* ((lines (split-string raw "\n"))
               (rows (mapcar (lambda (line) (split-string line "\t")) (cdr lines)))
               (rows (sort rows (lambda (a b)
                                  (let ((da (string-to-number (nth 0 a)))
                                        (db (string-to-number (nth 0 b))))
                                    (or (< da db)
                                        (and (= da db)
                                             (< (string-to-number (nth 1 a))
                                                (string-to-number (nth 1 b))))))))))
          (when rows
            (cons (car lines)
                  (mapcar (lambda (r) (list (nth 2 r) (or (nth 3 r) "") (nth 4 r)))
                          rows)))))))

  (defun apple-music-album-select ()
    "Choose a track from the album of the current track and play it."
    (interactive)
    (let ((data (apple-music--album-tracks)))
      (if (null data)
          (message "[apple-music] No current album (is Music.app playing?)")
        (let* ((rows (cdr data))
               (now (apple-music--applescript "persistent ID of current track"))
               (cur (or (cl-position now rows :key #'caddr :test #'equal) -1))
               (idx (apple-music--choose-track
                     (format "Album [%s]: " (car data))
                     (mapcar (lambda (r) (cons (car r) (cadr r))) rows)
                     (1+ cur))))
          (when idx
            (apple-music--applescript
             (format "play (first track of library playlist 1 whose persistent ID is \"%s\")"
                     (apple-music--quote (caddr (nth (1- idx) rows))))
             t)
            (message "♫ Playing: %s" (car (nth (1- idx) rows))))))))

    "Run osascript with LINES asynchronously and call CALLBACK with the output.

                         (kill-buffer buf)

:CUSTOM_ID: dev-run
:header-args:emacs-lisp: :tangle ../lisp/dev/dev-run.el
:END:

コードを書いたあとの「動かす・確かめる」を受け持つモジュールです。デバッガ、プロジェクトごとの環境、テストの実行の 3 つを置きます。

| 機能 | 中身 | キー |
|---+---+---|
| デバッガ | =dape= （DAP クライアント。デバッグアダプタは別に入れる） | =M-SPC c D= で開始、 =M-SPC c B= でブレークポイント |
| 環境 | =envrc= （direnv の環境をバッファごとに適用） | なし（ =direnv= があるときだけ自動で有効） |
| テスト | =dev-run-test= 。プロジェクトの種類からコマンドを決めて =compile= で走らせる | =M-SPC c t= で実行、 =M-SPC c T= で再実行 |

*テストのコマンドの決め方*

プロジェクトの根（ =project.el= の根。無ければ今のディレクトリ）にあるファイルで決めます。上から順に最初に当たったものを使い、
どれにも当たらなければ、コマンドを尋ねます。

| ファイル | コマンド |
|---+---|
| =Makefile= （ =test:= の目標がある場合のみ） | =make test= |
| =package.json= | =npm test= |
| =go.mod= | =go test ./...= |
| =Cargo.toml= | =cargo test= |
| =pyproject.toml= または =pytest.ini= | =python -m pytest= |

一覧は =dev-run-test-commands= で変えられます。

*envrc を条件つきで有効にする理由*

=envrc-global-mode= は、 =direnv= が無い環境でも有効にすると、バッファを開くたびに警告を出します。
そのため =direnv= の実行ファイルがあるときだけ、 =after-init-hook= で有効にします。

#+begin_src emacs-lisp
  ;;; dev/dev-run.el --- 実行と検証: デバッガ・環境・テスト -*- lexical-binding: t; -*-
  ;;
  ;; Copyright (c) 2021-2026
  ;; Author: YAMASHITA, Takao
  ;; License: GNU GPL v3 or later
  ;;
  ;; Category: dev
  ;;
  ;;; Commentary:
  ;;
  ;; コードを動かして確かめるための補助。
  ;;
  ;; 本ファイルの責務:
  ;; - dape（DAP クライアント）を導入する。デバッグアダプタ（debugpy、lldb-dap、delve など）は
  ;;   別に入れる必要がある
  ;; - envrc を、direnv があるときだけ有効にする
  ;; - `dev-run-test' / `dev-run-test-rerun': プロジェクトの種類からテストのコマンドを決め、
  ;;   `compile' で走らせる
  ;; - キーを `ui-leader-c-map' に割り当てる（M-SPC c D / B / t / T）。キーは
  ;;   `after-init-hook'（depth 90）で割り当てる。マップは ui-keymap の `after-init-hook'
  ;;   （depth 0）で作られるため、その後に走らせる必要がある
  ;;; Code:

  (eval-when-compile (require 'leaf) (require 'cl-lib) (require 'subr-x))

  ;; バイトコンパイル時の警告を避けるための宣言（実体は各パッケージ／ライブラリ側にある）。
  (defvar ui-leader-c-map)
  (declare-function project-current "project")
  (declare-function project-root "project")
  (declare-function envrc-global-mode "envrc")
  (declare-function dape "dape")
  (declare-function dape-breakpoint-toggle "dape")

  (defgroup dev-run nil
    "Debugger, environment and test running."
    :group 'tools)

  (defcustom dev-run-test-commands
    '(("Makefile"       . "make test")
      ("package.json"   . "npm test")
      ("go.mod"         . "go test ./...")
      ("Cargo.toml"     . "cargo test")
      ("pyproject.toml" . "python -m pytest")
      ("pytest.ini"     . "python -m pytest"))
    "Project files and the test command each one selects, tried from the top.
  A Makefile only counts when it has a `test:' target."
    :type '(alist :key-type string :value-type string)
    :group 'dev-run)

  (defvar dev-run--last-test nil
    "The directory and command of the last test run, as (DIR . COMMAND), or nil.")

  (defun dev-run--makefile-has-test-p (dir)
    "Return non-nil when the Makefile in DIR has a `test:' target."
    (with-temp-buffer
      (insert-file-contents (expand-file-name "Makefile" dir))
      (goto-char (point-min))
      (re-search-forward "^test[ \t]*:" nil t)))

  (defun dev-run-test-command (dir)
    "Return the test command for the project in DIR, or nil when none is known."
    (let (found)
      (dolist (entry dev-run-test-commands)
        (when (and (not found)
                   (file-exists-p (expand-file-name (car entry) dir))
                   (or (not (equal (car entry) "Makefile"))
                       (dev-run--makefile-has-test-p dir)))
          (setq found (cdr entry))))
      found))

  (defun dev-run--root ()
    "Return the project root, or the current directory outside a project."
    (if-let* ((project (project-current)))
        (project-root project)
      default-directory))

  ;;;###autoload
  (defun dev-run-test ()
    "Run the project's tests with `compile'.
  The command comes from `dev-run-test-commands'; ask when none matches."
    (interactive)
    (let* ((root (dev-run--root))
           (command (or (dev-run-test-command root)
                        (read-shell-command "テストのコマンド: ")))
           (default-directory root))
      (setq dev-run--last-test (cons root command))
      (compile command)))

  ;;;###autoload
  (defun dev-run-test-rerun ()
    "Run the last test command again, or `dev-run-test' when there is none."
    (interactive)
    (if dev-run--last-test
        (let ((default-directory (car dev-run--last-test)))
          (compile (cdr dev-run--last-test)))
      (dev-run-test)))

  (defun dev-run--enable-envrc ()
    "Enable `envrc-global-mode' when direnv is installed."
    (when (executable-find "direnv")
      (envrc-global-mode 1)))

  (defun dev-run--bind-keys ()
    "Bind the run and debug commands in `ui-leader-c-map' when it exists."
    (when (and (boundp 'ui-leader-c-map) (keymapp ui-leader-c-map))
      (keymap-set ui-leader-c-map "t" #'dev-run-test)
      (keymap-set ui-leader-c-map "T" #'dev-run-test-rerun)
      (keymap-set ui-leader-c-map "D" #'dape)
      (keymap-set ui-leader-c-map "B" #'dape-breakpoint-toggle)))

  ;; depth 90: ui-keymap--startup-init（depth 0）がマップを作った後に走らせる。
  (add-hook 'after-init-hook #'dev-run--bind-keys 90)

  (leaf dape
    :straight t
    :commands (dape dape-breakpoint-toggle dape-quit)
    :custom ((dape-buffer-window-arrangement . 'right)))

  (leaf envrc
    :straight t
    :commands (envrc-global-mode)
    :hook (after-init-hook . dev-run--enable-envrc))

  (provide 'dev-run)
  ;;; dev/dev-run.el ends here
#+end_src
